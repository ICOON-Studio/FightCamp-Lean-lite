1. FootLockPlantModifier.gd

Enforces C223 Gate Invariant 2: Maximum foot sliding distance $d_{\text{slide}} \le 1.5\text{ cm}$ for support feet while planted during strike execution and defensive transitions.

class_name FootLockPlantModifier
extends SkeletonModifier3D

@export var enable_foot_lock: bool = false # Rule A: Off by default
@export var max_allowable_slide_m: float = 0.015 # 1.5 cm gate ceiling

# Telemetry
@export var measured_left_slide_cm: float = 0.0
@export var measured_right_slide_cm: float = 0.0

var _skeleton: Skeleton3D = null
var _l_foot_idx: int = -1
var _r_foot_idx: int = -1

var _l_planted: bool = false
var _r_planted: bool = false

var _l_anchor_pos: Vector3 = Vector3.ZERO
var _r_anchor_pos: Vector3 = Vector3.ZERO
var _is_initialized: bool = false

func _init() -> void:
	active = true

func set_foot_planted(is_left: bool, planted: bool) -> void:
	if not enable_foot_lock:
		return
	if is_left:
		if planted and not _l_planted and _skeleton != null and _l_foot_idx >= 0:
			_l_anchor_pos = _skeleton.get_bone_global_pose(_l_foot_idx).origin
		_l_planted = planted
	else:
		if planted and not _r_planted and _skeleton != null and _r_foot_idx >= 0:
			_r_anchor_pos = _skeleton.get_bone_global_pose(_r_foot_idx).origin
		_r_planted = planted

func _validate_setup() -> void:
	_skeleton = get_skeleton()
	if _skeleton == null:
		return
	_l_foot_idx = _skeleton.find_bone("mixamorig_LeftFoot")
	_r_foot_idx = _skeleton.find_bone("mixamorig_RightFoot")
	_is_initialized = (_l_foot_idx >= 0 and _r_foot_idx >= 0)

func _process_modification() -> void:
	if not enable_foot_lock:
		measured_left_slide_cm = 0.0
		measured_right_slide_cm = 0.0
		return

	if not _is_initialized:
		_validate_setup()
		if not _is_initialized:
			return

	# Left foot constraint
	if _l_planted:
		var current_pose: Transform3D = _skeleton.get_bone_global_pose(_l_foot_idx)
		var drift: Vector3 = current_pose.origin - _l_anchor_pos
		drift.y = 0.0 # Preserve vertical ankle compliance
		measured_left_slide_cm = drift.length() * 100.0
		if drift.length() > 0.0001:
			var clamped_origin: Vector3 = _l_anchor_pos + drift.normalized() * minf(drift.length(), max_allowable_slide_m)
			clamped_origin.y = current_pose.origin.y
			current_pose.origin = clamped_origin
			_skeleton.set_bone_global_pose(_l_foot_idx, current_pose)

	# Right foot constraint
	if _r_planted:
		var current_pose: Transform3D = _skeleton.get_bone_global_pose(_r_foot_idx)
		var drift: Vector3 = current_pose.origin - _r_anchor_pos
		drift.y = 0.0
		measured_right_slide_cm = drift.length() * 100.0
		if drift.length() > 0.0001:
			var clamped_origin: Vector3 = _r_anchor_pos + drift.normalized() * minf(drift.length(), max_allowable_slide_m)
			clamped_origin.y = current_pose.origin.y
			current_pose.origin = clamped_origin
			_skeleton.set_bone_global_pose(_r_foot_idx, current_pose)

2. PunchTimeWarpController.gd

Applies normalized Hermite S-Curve time-warping to punch trajectories so velocity snaps cleanly to impact without altering raw clip keyframe timing.

class_name PunchTimeWarpController
extends Node

@export var enable_time_warp: bool = false # Rule A
@export var current_warped_time: float = 0.0

# Mathematical Hermite constants
# Warping formula: S(tau) = 3*tau^2 - 2*tau^3 with boundary locking: S(0)=0, S(tau_c)=tau_c, S(1)=1
var _clip_length: float = 0.30
var _contact_time: float = 0.125
var _is_warping: bool = false

func start_strike(clip_length_sec: float, certified_contact_sec: float) -> void:
	_clip_length = maxf(clip_length_sec, 0.05)
	_contact_time = clampf(certified_contact_sec, 0.0, _clip_length)
	_is_warping = enable_time_warp

func sample_time(raw_elapsed_sec: float) -> float:
	if not enable_time_warp or not _is_warping:
		current_warped_time = raw_elapsed_sec
		return raw_elapsed_sec

	var tau: float = clampf(raw_elapsed_sec / _clip_length, 0.0, 1.0)
	var tau_c: float = _contact_time / _clip_length

	var warped_tau: float = 0.0
	if tau <= tau_c and tau_c > 0.0:
		# Acceleration phase: smooth start into locked contact frame
		var local_ratio: float = tau / tau_c
		var hermite: float = (3.0 * pow(local_ratio, 2.0)) - (2.0 * pow(local_ratio, 3.0))
		warped_tau = hermite * tau_c
	else:
		# Retraction phase: snap return from contact to neutral stance
		var local_ratio: float = (tau - tau_c) / (1.0 - tau_c)
		var hermite: float = (3.0 * pow(local_ratio, 2.0)) - (2.0 * pow(local_ratio, 3.0))
		warped_tau = tau_c + (hermite * (1.0 - tau_c))

	current_warped_time = warped_tau * _clip_length
	return current_warped_time

3. HeadImpactModifier.gd

Applies angular head deflection upon clean impact or partial guard penetration, scaling dynamically with strike kinetic energy.

class_name HeadImpactModifier
extends SkeletonModifier3D

@export var enable_head_impact: bool = false # Rule A
@export var measured_deflection_deg: float = 0.0

const RESTORATION_STIFFNESS: float = 380.0 # Rad/s^2 equivalent
const IMPACT_DAMPING: float = 24.5

var _skeleton: Skeleton3D = null
var _head_idx: int = -1
var _neck_idx: int = -1
var _current_rot_offset: Quaternion = Quaternion.IDENTITY
var _rot_velocity: Vector3 = Vector3.ZERO
var _is_initialized: bool = false

func _init() -> void:
	active = true

func apply_impact(direction_vector: Vector3, damage_amount: float) -> void:
	if not enable_head_impact:
		return
	var impulse_magnitude: float = clampf(damage_amount * 0.045, 0.2, 1.8)
	var torque_axis: Vector3 = direction_vector.cross(Vector3.UP).normalized()
	_rot_velocity += torque_axis * impulse_magnitude * 22.0

func _validate_setup() -> void:
	_skeleton = get_skeleton()
	if _skeleton == null:
		return
	_head_idx = _skeleton.find_bone("mixamorig_Head")
	_neck_idx = _skeleton.find_bone("mixamorig_Neck")
	_is_initialized = (_head_idx >= 0 and _neck_idx >= 0)

func _process_modification() -> void:
	if not enable_head_impact:
		measured_deflection_deg = 0.0
		return

	if not _is_initialized:
		_validate_setup()
		if not _is_initialized:
			return

	var dt: float = clampf(get_physics_process_delta_time(), 0.001, 0.033)

	# Spring-damper integration on rotation
	var euler: Vector3 = _current_rot_offset.get_euler()
	var spring_acc: Vector3 = (-RESTORATION_STIFFNESS * euler) - (IMPACT_DAMPING * _rot_velocity)
	_rot_velocity += spring_acc * dt
	euler += _rot_velocity * dt
	_current_rot_offset = Quaternion.from_euler(euler)

	measured_deflection_deg = rad_to_deg(_current_rot_offset.get_angle())

	var base_head_rot: Quaternion = _skeleton.get_bone_pose_rotation(_head_idx)
	_skeleton.set_bone_pose_rotation(_head_idx, base_head_rot * _current_rot_offset)

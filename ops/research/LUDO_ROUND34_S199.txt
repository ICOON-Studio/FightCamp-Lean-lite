1.2 The Authoritative Recommendation: The Wall-Mounted Cantilevered I-Beam Track (System 2 & 4 Hybrid)

We select the Industrial Wall-Mounted Cantilever Box-Track & I-Beam Spine.

In commercial boxing gyms, heavy bags are almost never hung from the roof joists (which would fatigue roof engineering stamps). Instead, they hang from a heavy structural steel W-section (I-beam) cantilevered off the perimeter load-bearing wall, stabilized with high-tensile 45° diagonal square-hollow-section (SHS) kicker struts.

This anchors the visual weight to the solid brick wall (which is drawn at 3.6m height), keeping the floor open for footwork decals, shadow projections, and pathfinding navmeshes.

1.3 The Physics Pendulum Rigging & Mathematical Model

The bag acts as a physical pendulum rotating around an exact, stable pivot point. Because the overhead rail is rendered, the attachment point is visible, grounding the movement.

# HeavyBagPhysicsRig.gd — Physics Pendulum Simulation (Godot 4.7.2 Forward+)
class_name HeavyBagPhysicsRig extends Node3D

@export var bag_mass_kg: float = 45.0          # 100 lb heavy bag
@export var bag_length_m: float = 1.35          # Cylinder height
@export var chain_length_m: float = 0.45        # Shackle to bag rim
@export var natural_frequency_rad: float = 2.55 # w = sqrt(g / L_eff)
@export var damping_coefficient: float = 0.88   # Air and chain friction decay
@export var max_angular_displacement: float = 0.42 # Max swing radians (~24°)

# PIVOT POINT DEFINITION:
# World-space position of the shackle eyelet beneath the cantilever I-beam.
# Local space relative to bag station root: Vector3(0.0, 2.15, 0.0)
@onready var pivot_point: Marker3D = $ShacklePivot
@onready var bag_mesh_node: Node3D = $ShacklePivot/BagCylinder
@onready var chain_mesh_node: Node3D = $ShacklePivot/ChainAssembly

var angular_velocity: Vector2 = Vector2.ZERO # X = pitch swing, Y = roll swing
var angular_displacement: Vector2 = Vector2.ZERO

func _physics_process(delta: float) -> void:
    if angular_displacement.length_squared() > 0.00001 or angular_velocity.length_squared() > 0.00001:
        # Pendulum restoring torque: T = - (g / L) * sin(theta)
        var restoring_accel_x: float = -pow(natural_frequency_rad, 2.0) * sin(angular_displacement.x)
        var restoring_accel_y: float = -pow(natural_frequency_rad, 2.0) * sin(angular_displacement.y)
        
        # Integration with drag
        angular_velocity.x += restoring_accel_x * delta
        angular_velocity.y += restoring_accel_y * delta
        angular_velocity *= exp(-damping_coefficient * delta)
        
        angular_displacement.x += angular_velocity.x * delta
        angular_displacement.y += angular_velocity.y * delta
        
        # Apply transformation to the pivot marker
        pivot_point.rotation.x = angular_displacement.x
        pivot_point.rotation.z = angular_displacement.y
    else:
        # Subtle idle air-draft micro-sway
        var t: float = Time.get_ticks_msec() * 0.001
        pivot_point.rotation.x = sin(t * 0.75) * 0.006
        pivot_point.rotation.z = cos(t * 0.55) * 0.004

func apply_punch_impact(impulse_joules: float, hit_direction: Vector3, contact_height_m: float) -> void:
    # contact_height_m: local height above floor (default impact sweet-spot is 0.95m)
    var effective_lever_arm: float = (2.15 - contact_height_m) # Distance from pivot to glove contact
    var angular_impulse: float = (impulse_joules / (bag_mass_kg * effective_lever_arm)) * 0.08
    
    # Project impact direction into local pendulum pitch and roll
    angular_velocity.x += hit_direction.z * angular_impulse
    angular_velocity.y += -hit_direction.x * angular_impulse
    
    # Clamp to prevent unnatural clipping
    angular_velocity.x = clampf(angular_velocity.x, -max_angular_displacement, max_angular_displacement)
    angular_velocity.y = clampf(angular_velocity.y, -max_angular_displacement, max_angular_displacement)

// CantileverOcclusionFade.gdshader (Godot 4.7.2 Forward+)
shader_type spatial;
render_mode blend_mix, depth_draw_opaque, cull_back;

uniform sampler2D texture_albedo : source_color, filter_linear_mipmap;
uniform sampler2D texture_roughness : hint_roughness_gray;
uniform sampler2D texture_metallic : hint_default_white;
uniform sampler2D texture_normal : hint_normal;

uniform float fade_distance_min = 2.2; // Start fading out
uniform float fade_distance_max = 3.8; // Fully opaque

void fragment() {
    vec4 albedo_tex = texture(texture_albedo, UV);
    ALBEDO = albedo_tex.rgb;
    METALLIC = texture(texture_metallic, UV).r;
    ROUGHNESS = texture(texture_roughness, UV).r;
    NORMAL_MAP = texture(texture_normal, UV).rgb;
    
    // Compute distance from camera lens to current fragment
    float dist = length(VERTEX);
    float alpha_factor = clamp((dist - fade_distance_min) / (fade_distance_max - fade_distance_min), 0.0, 1.0);
    ALPHA = albedo_tex.a * alpha_factor;
}

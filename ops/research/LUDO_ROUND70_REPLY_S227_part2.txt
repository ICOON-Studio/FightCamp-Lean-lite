Creates an authentic chamfered industrial button matching Gleason specifications

static func create_chamfer_button(label_text: String, is_primary: bool = false, custom_h: int = 36) -> Button:
var btn := Button.new()
btn.text = label_text.to_upper()
btn.custom_minimum_size = Vector2(0, custom_h)
btn.mouse_default_cursor_shape = Control.CURSOR_POINTING_HAND

if is_primary:
	btn.theme_type_variation = "ButtonPrimary"
	btn.add_theme_font_size_override("font_size", FONT_SIZE_SUBTITLE)
else:
	btn.theme_type_variation = "ButtonSecondary"
	btn.add_theme_font_size_override("font_size", FONT_SIZE_SUBHEAD)

# Micro physical hover tween
btn.mouse_entered.connect(func():
	if not btn.disabled and btn.is_inside_tree():
		var tw := btn.create_tween()
		tw.tween_property(btn, "scale", Vector2(1.015, 1.015), DURATION_HOVER).set_trans(Tween.TRANS_SINE)
)
btn.mouse_exited.connect(func():
	if btn.is_inside_tree():
		var tw := btn.create_tween()
		tw.tween_property(btn, "scale", Vector2.ONE, DURATION_HOVER).set_trans(Tween.TRANS_SINE)
)
btn.button_down.connect(func():
	btn.position.y += 1.0
)
btn.button_up.connect(func():
	btn.position.y -= 1.0
)
return btn

Creates a physical, multi-segmented meter (pips) for core boxer attributes

static func create_segmented_gauge(pip_count: int = 10, current_val: int = 50, bar_h: int = 12) -> HBoxContainer:
var container := HBoxContainer.new()
container.custom_minimum_size = Vector2(0, bar_h)
container.add_theme_constant_override("separation", SP_2)

var active_pips := int(round(float(clampi(current_val, 0, 100)) / (100.0 / float(pip_count))))

for i in range(pip_count):
	var pip := ColorRect.new()
	pip.size_flags_horizontal = Control.SIZE_EXPAND_FILL
	pip.size_flags_vertical = Control.SIZE_EXPAND_FILL
	if i < active_pips:
		if current_val >= 90:
			pip.color = COLOR_GOLD_BRIGHT
		elif current_val >= 75:
			pip.color = COLOR_BRASS_ANTIQUE
		elif current_val <= 35:
			pip.color = COLOR_OXBLOOD_BRIGHT
		else:
			pip.color = COLOR_GOLD_MUTED
	else:
		pip.color = COLOR_BG_RECESS
	container.add_child(pip)

return container

	"scrim": scrim,
	"modal": modal,
	"body": body,
	"footer": footer,
	"close_button": close_btn
}

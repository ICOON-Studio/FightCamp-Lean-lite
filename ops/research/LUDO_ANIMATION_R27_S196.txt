3. JUDO & BJJ SPECIFICATIONS (TWO-FIGHTER COUPLED SYSTEMS)
3.1 Kinematic Interaction Architecture & Pipeline

Grappling cannot be rendered as two desynchronized, independent skeletal clips playing next to each other. It must be executed as a unified, dual-skeleton interaction system bounded by three physical invariants:

[20 Hz Discrete Sim Engine]
       │
       ▼
[FightBeatAdapter: Emits (GripState, KuzushiVector, TransitionPhase)]
       │
       ▼
[DualSkeletonConstraintNode: Locks Tori Grip Bones <-> Uke Socket Points]
       │
       ▼
[FightPoseSolver: Solves Coupled COM -> Pulls Uke into Tori Base of Support]
       │
       ▼
[Flight & Mat Impact Release: Uncouples Uke at Apex -> Solves Ukemi Breakfall]

Grip Sockets (Transform Anchors): Rigid spatial transforms between Tori's hands and Uke's gi/anatomical anchor points.
Coupled Center of Mass: Tori cannot elevate or pivot Uke without first pulling Uke's projected horizontal center of mass ($COM_{xy}$) inside or across Tori's Base of Support ($BOS$).
Paired Kinetic Timelines: Uke does not play an autonomous "reaction" clip. Uke's root and pelvic trajectory are procedurally derived from Tori's hips until the instant of physical separation (the throw apex), where Uke transitions to a breakfall state (Ukemi).

Correction 1: Failure Mode 2 — Persistent Velocity Vector State for Inertial Hit Impulses

get_bone_pose_position returns a local spatial offset, not a velocity. Physics-based impulses require maintaining a persistent velocity state that updates each frame.

# File: FightStagePlayer.gd
# Corrected: Maintains explicit linear velocity states and processes in deferred bone updates
var bone_velocity_state: Dictionary = {} # StringName -> Vector3

func apply_inertial_hit_impulse(skeleton: Skeleton3D, bone_name: StringName, impulse_newtons: Vector3, mass_kg: float) -> void:
	var delta_v: Vector3 = impulse_newtons / maxf(mass_kg, 1.0)
	if not bone_velocity_state.has(bone_name):
		bone_velocity_state[bone_name] = Vector3.ZERO
	bone_velocity_state[bone_name] += delta_v

func process_inertial_velocity_integration(skeleton: Skeleton3D, delta: float) -> void:
	for bone_name in bone_velocity_state.keys():
		var b_idx: int = skeleton.find_bone(bone_name)
		if b_idx == -1:
			continue
		var current_vel: Vector3 = bone_velocity_state[bone_name]
		var cur_pose_pos: Vector3 = skeleton.get_bone_pose_position(b_idx)
		# Integrate velocity into position offset
		cur_pose_pos += current_vel * delta
		# Apply drag damping
		bone_velocity_state[bone_name] = current_vel.move_toward(Vector3.ZERO, 9.8 * 2.0 * delta)
		skeleton.set_bone_pose_position(b_idx, cur_pose_pos)

func solve_cervical_chain_deferred(skeleton: Skeleton3D, target_aim_point: Vector3, chin_tuck_factor: float) -> void:
	var head_idx: int = skeleton.find_bone("Head")
	var neck_idx: int = skeleton.find_bone("Neck")
	var spine2_idx: int = skeleton.find_bone("Spine2")
	if head_idx == -1 or neck_idx == -1 or spine2_idx == -1:
		return
		
	var head_global: Transform3D = skeleton.get_bone_global_pose(head_idx)
	var to_aim: Vector3 = (target_aim_point - head_global.origin).normalized()
	var forward_vec: Vector3 = -head_global.basis.z # -Z forward convention
	
	var full_rotation: Quaternion = Quaternion(forward_vec, to_aim)
	var limited_rot: Quaternion = Quaternion.IDENTITY.slerp(full_rotation, 0.45)
	
	# Distribute rotation down the spine
	skeleton.set_bone_pose_rotation(spine2_idx, skeleton.get_bone_pose_rotation(spine2_idx) * limited_rot.slerp(Quaternion.IDENTITY, 0.7))
	skeleton.set_bone_pose_rotation(neck_idx, skeleton.get_bone_pose_rotation(neck_idx) * limited_rot.slerp(Quaternion.IDENTITY, 0.4))
	
	# Clamp chin downward to protect the neck
	var tuck_quat: Quaternion = Quaternion(Vector3.RIGHT, deg_to_rad(14.0 * clampf(chin_tuck_factor, 0.0, 1.0)))
	skeleton.set_bone_pose_rotation(head_idx, (skeleton.get_bone_pose_rotation(head_idx) * limited_rot * tuck_quat).normalized())

    def __init__(self, csv_filepath: str):

        self.failures = []

    def verify_foot_sliding(self, max_slide_speed_mps: float = 0.025) -> bool:
        """Asserts that planted feet do not slide across the floor canvas."""
        for foot in ['l', 'r']:
            planted = self.df[f'{foot}_foot_planted'] == 1
            px = self.df.loc[planted, f'{foot}_ankle_px'].to_numpy()
            pz = self.df.loc[planted, f'{foot}_ankle_pz'].to_numpy()
            if len(px) < 2:
                continue

            vz = np.diff(pz) / self.dt
            horizontal_speed = np.sqrt(vx**2 + vz**2)
            max_speed = np.max(horizontal_speed) if len(horizontal_speed) > 0 else 0.0
            if max_speed > max_slide_speed_mps:
                self.failures.append(f"FAIL: {foot.upper()} foot planted slide speed: {max_speed*100:.2f} cm/s (Max: {max_slide_speed_mps*100:.2f} cm/s)")

FightCamp · UI Census Row 6: The Schedule & The Cards (Screen Audit & Spec)

In accordance with the Round 70 UI Bible design tokens (Deep Void #0d0c0b, Panel #1a1816, Surface Raised #22201d, Surface High #282521, Brass Border #4a3e2c, Championship Gold #f0c233, Cream Text #f5ebd7, Muted Sand #9e9484, 4 px corner radii, 36 px header tabs), here is the complete audit, layout architecture, comparative study, task build list, and Godot 4.7 GDScript implementation.

A. Visual & UX Defect Catalog
Picture 1: Tab "MY GYM" (Calendar & Agenda View)
Tab Active State Violation: The active tab MY GYM uses a teal/cyan highlight color (#22a699 / #00b4d8) instead of the Round-70 primary brand token Championship Gold (#f0c233) with a 2 px gold bottom indicator.
Cavernous Void / Empty Space Pathology: Below the lone monthly bill row (SUN 1 FEB), over 650 vertical pixels are dead empty space. There is no multi-week planner strip, no projected cash burn trajectory, no camp timeline bar, and no upcoming sparring milestones.
Floating "NEXT UP" Hero Card Misplacement: The banner "Sign your first fighter" floats awkwardly with disproportionate margins (y: 96 to y: 172), taking up 76 vertical pixels without integrating into the calendar time axis.
Isolated Right Rail ("YOUR FIGHTERS"): The right sidebar reads "NO FIGHTERS YET - SIGN ONE IN SCOUTING" in a massive panel (w: ~440px, h: ~780px) with zero utility, missing onboarding shortcuts, prospect previews, or roster stat summaries.
Clipped & Overlapping Guide Pill: The yellow/gold banner in the bottom-right corner ("GUIDE · Sign your first boxer") overlaps the border frame edge and interferes with primary navigation anchors.
Orphaned Row Action: The REVIEW button on the SUN 1 FEB bill item lacks cost breakdown hierarchy, projected balance impact, or tooltip expansion.
Missing Date Progression Granularity: No day-of-week calendar grid, 14-day rolling view, or visual demarcation between Camp Weeks, Fight Weeks, and Purse Settlement dates.
Picture 2: Tab "THE CARDS" (World Event Browser)
Hardcoded Disciplines ("DISC. = HYBRID"): Every single card row displays HYBRID as its discipline, including boxing, taekwondo, wrestling, and judo federations.
Severe Pill Label Truncation (Red Flag Pill): The right status pill cuts off text abruptly: "NO ROSTER FIGHTER MATCHES THIS DIVIS...". It exceeds its pill bounding box without ellipsis or wrapping, breaking container bounds.
Completely Empty "LEVEL" Column: The LEVEL data column has zero rendered content across all 11 visible rows. It should display tournament/card tier indicators (Regional C, National B, Continental A, World Title).
Truncated Organization Names: ORG entries like UNITED KINGDOM COMBAT ASSOCIATION 1 and UNIVERSAL REALITY COMBAT CHAMPIONSHIP 1 run into the adjacent DISC. column due to static column bounds without proper text clipping (TextServer autowrap or clip margin).
Dead Footer Buttons: OPEN EVENT and CANNOT BOOK are rendered in disabled slate grey (#282521 with #524c42 text) with zero tooltips explaining why they are locked (e.g., "Requires an active licensed boxer in Welterweight (147 lbs)").
Sub-Navigation Filter Tag Inconsistency: Filter pills (ALL (11), MY FIGHTER DISC., MY REGION, MY ORG, BOOKABLE) lack active visual states, badge counts, and clear toggle cues.
Row Density & Visual Hierarchy Breakdown: Rows use flat zebra striping with washed-out borders; there is no hover elevation (#22201d -> #282521), no left accent pip indicating urgency, and no weight-class callout.

1. Status Pill Component (ui/components/StatusPill.gd)
class_name StatusPill
extends PanelContainer

enum StatusType {
	ELIGIBLE,
	NO_WEIGHT_MATCH,
	NO_RANK_MATCH,
	CARD_FULL,
	CONTRACT_EXPIRED
}

@onready var label: Label = $Label

const STYLE_MAP := {
	StatusType.ELIGIBLE: {
		"bg": Color("#1f3821"),
		"border": Color("#3e7b44"),
		"text": Color("#7bd685"),
		"label": "ELIGIBLE FOR BOUT"
	},
	StatusType.NO_WEIGHT_MATCH: {
		"bg": Color("#38201f"),
		"border": Color("#7b3e3e"),
		"text": Color("#e58282"),
		"label": "NO ROSTER FIT (WEIGHT)"
	},
	StatusType.NO_RANK_MATCH: {
		"bg": Color("#382e18"),
		"border": Color("#7b6326"),
		"text": Color("#f0c233"),
		"label": "RANK REQUIREMENT UNMET"
	},
	StatusType.CARD_FULL: {
		"bg": Color("#201e1c"),
		"border": Color("#4a3e2c"),
		"text": Color("#9e9484"),
		"label": "NO SLOTS REMAINING"
	},
	StatusType.CONTRACT_EXPIRED: {
		"bg": Color("#2b1d1d"),
		"border": Color("#542828"),
		"text": Color("#c47878"),
		"label": "PROMOTER RESTRICTED"
	}
}

func _ready() -> void:
	custom_minimum_size = Vector2(210, 26)
	size_flags_horizontal = Control.SIZE_SHRINK_END
	size_flags_vertical = Control.SIZE_SHRINK_CENTER

func setup(status: StatusType, custom_override: String = "") -> void:
	var cfg = STYLE_MAP.get(status, STYLE_MAP[StatusType.CARD_FULL])
	
	var sb := StyleBoxFlat.new()
	sb.bg_color = cfg["bg"]
	sb.border_color = cfg["border"]
	sb.set_border_width_all(1)
	sb.set_corner_radius_all(4)
	sb.content_margin_left = 8
	sb.content_margin_right = 8
	sb.content_margin_top = 2
	sb.content_margin_bottom = 2
	add_theme_stylebox_override("panel", sb)
	
	label.text = custom_override if custom_override != "" else cfg["label"]
	label.add_theme_color_override("font_color", cfg["text"])
	label.add_theme_font_size_override("font_size", 11)
	label.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	label.vertical_alignment = VERTICAL_ALIGNMENT_CENTER
	label.text_overrun_behavior = TextServer.OVERRUN_TRIM_ELLIPSIS

2. World Card Table Row (ui/calendar/WorldCardRow.gd)
class_name WorldCardRow
extends PanelContainer

signal card_selected(card_data: Dictionary)
signal booking_requested(card_data: Dictionary)

@export var token_normal_bg := Color("#1a1816")
@export var token_hover_bg := Color("#22201d")
@export var token_selected_bg := Color("#282521")
@export var token_border := Color("#4a3e2c")
@export var token_gold := Color("#f0c233")
@export var token_cream := Color("#f5ebd7")
@export var token_sand := Color("#9e9484")

var _card_data: Dictionary = {}
var _is_selected: bool = false
var _style_box: StyleBoxFlat

@onready var lbl_date: Label = %LblDate
@onready var lbl_day: Label = %LblDay
@onready var lbl_event: Label = %LblEvent
@onready var lbl_disc: Label = %LblDiscipline
@onready var lbl_level: Label = %LblLevel
@onready var lbl_venue: Label = %LblVenue
@onready var lbl_purse: Label = %LblPurse
@onready var status_pill: StatusPill = %StatusPill

func _ready() -> void:
	custom_minimum_size = Vector2(1380, 52)
	mouse_entered.connect(_on_mouse_enter)
	mouse_exited.connect(_on_mouse_exit)
	gui_input.connect(_on_gui_input)
	_setup_base_style()

func populate(data: Dictionary) -> void:
	_card_data = data
	lbl_date.text = data.get("date_short", "01 JAN")
	lbl_day.text = data.get("day_name", "SAT")
	lbl_event.text = data.get("name", "Unknown Promotion")
	lbl_disc.text = data.get("discipline", "BOXING").to_upper()
	lbl_level.text = data.get("tier", "REG-B")
	lbl_venue.text = "%s, %s" % [data.get("venue", "Arena"), data.get("city", "Local")]
	
	var purse_val: int = data.get("base_purse", 0)
	lbl_purse.text = "$%s" % String.num(purse_val, 0)
	
	# Determine Status
	var status: StatusPill.StatusType = data.get("status_code", StatusPill.StatusType.CARD_FULL)
	status_pill.setup(status)
	_update_style()

func set_selected(selected: bool) -> void:
	_is_selected = selected
	_update_style()

func _setup_base_style() -> void:
	_style_box = StyleBoxFlat.new()
	_style_box.set_corner_radius_all(4)
	_style_box.set_border_width_all(1)
	add_theme_stylebox_override("panel", _style_box)
	_update_style()

func _update_style() -> void:
	if _is_selected:
		_style_box.bg_color = token_selected_bg
		_style_box.border_color = token_gold
		_style_box.border_width_left = 4
	else:
		_style_box.bg_color = token_normal_bg
		_style_box.border_color = token_border
		_style_box.border_width_left = 1

func _on_mouse_enter() -> void:
	if not _is_selected:
		var tween = create_tween().set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_OUT)
		tween.tween_property(_style_box, "bg_color", token_hover_bg, 0.1)

func _on_mouse_exit() -> void:
	if not _is_selected:
		var tween = create_tween().set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_OUT)
		tween.tween_property(_style_box, "bg_color", token_normal_bg, 0.1)

func _on_gui_input(event: InputEvent) -> void:
	if event is InputEventMouseButton and event.button_index == MOUSE_BUTTON_LEFT and event.pressed:
		card_selected.emit(_card_data)

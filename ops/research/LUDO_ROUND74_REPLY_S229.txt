Next Bout Widget Disconnected from Roster State: Top-right "NO BOUT BOOKED / FIND A FIGHT >" container lacks a primary CTA button hierarchy (#f0c233 Championship Gold accent) and floats with bare right-padding instead of locking to the right margin grid (x=1896).

Left Rail Contrast & Hitbox Failure: The icon rail on the left uses ungrounded icons with sub-30% opacity grey glyphs without pill backing, hover plates, or active indicator notch (the Gold 2×24px active left pip required by the UI Bible).
Camera Zoom Control Floating Naked: The [+] zoom button at (x=48, y=72) is floating adrift near the left rail with no grouped UI docking, creating visual clutter against the gym ceiling geometry.
Modal Button Visual Hierarchy: The primary button "Open the gym" in the story modal uses solid yellow with black text, but lacks the Round-70 4px radius, bevel edge, and hover token state; the secondary button "LATER" has an irregular border thickness and floating baseline alignment.
Right Edge Calendar Tab Clipping & Readability: The "27d" calendar drawer tab at (x=1896, y=360) has micro-sized unreadable vertical text, an unstyled pill handle, and fails to align with the standard 48px grid units.
3D Floor Bleed Behind News Bar: The bottom marquee bar lacks a solid #1a1816 backing and gradient top-feather (1px #282521 border), allowing the grey gym asphalt and floor chalk lines to create visual noise beneath running text.

1. News Ticker Component (news_ticker.gd)
class_name NewsTicker
extends Control

@export var scroll_speed: float = 38.0
@export var feed_items: Array[String] = [
	"ANVIL STREET: Dessa stepped down as head trainer; gym ownership transferred.",
	"REGIONAL LEAGUE: Welterweight rankings updated for January cycle.",
	"EQUIPMENT: Heavy bag #3 leather strap inspected - optimal condition."
]

var _stream_label: Label
var _scroll_offset: float = 0.0
var _total_text_width: float = 0.0
var _is_paused: bool = false

const COLOR_VOID: Color = Color("#0d0c0b")
const COLOR_SURFACE: Color = Color("#121110")
const COLOR_BORDER: Color = Color("#22201d")
const COLOR_TEXT: Color = Color("#b5afa8")
const COLOR_GOLD: Color = Color("#f0c233")

func _ready() -> void:
	custom_minimum_size = Vector2(1920, 32)
	mouse_entered.connect(func(): _is_paused = true)
	mouse_exited.connect(func(): _is_paused = false)
	
	var bg := ColorRect.new()
	bg.color = COLOR_SURFACE
	bg.set_anchors_preset(PRESET_FULL_RECT)
	add_child(bg)
	
	var top_border := ColorRect.new()
	top_border.color = COLOR_BORDER
	top_border.size = Vector2(1920, 1)
	add_child(top_border)
	
	# Badge Tag
	var badge := PanelContainer.new()
	badge.size = Vector2(88, 32)
	var badge_style := StyleBoxFlat.new()
	badge_style.bg_color = Color("#1a1816")
	badge_style.border_width_right = 1
	badge_style.border_color = COLOR_BORDER
	badge.add_theme_stylebox_override("panel", badge_style)
	
	var badge_lbl := Label.new()
	badge_lbl.text = "FEED"
	badge_lbl.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	badge_lbl.vertical_alignment = VERTICAL_ALIGNMENT_CENTER
	badge_lbl.add_theme_color_override("font_color", COLOR_GOLD)
	badge_lbl.add_theme_font_size_override("font_size", 10)
	badge.add_child(badge_lbl)
	add_child(badge)
	
	# Scroll Container Node
	var clip_box := Control.new()
	clip_box.position = Vector2(96, 0)
	clip_box.size = Vector2(1824, 32)
	clip_box.clip_contents = true
	add_child(clip_box)
	
	_stream_label = Label.new()
	_stream_label.vertical_alignment = VERTICAL_ALIGNMENT_CENTER
	_stream_label.size = Vector2(0, 32)
	_stream_label.add_theme_color_override("font_color", COLOR_TEXT)
	_stream_label.add_theme_font_size_override("font_size", 12)
	clip_box.add_child(_stream_label)
	
	_rebuild_ticker_string()

func _rebuild_ticker_string() -> void:
	var separator: String = "    •    "
	var full_text: String = separator.join(feed_items) + separator
	_stream_label.text = full_text + full_text # Doubled for continuous looping
	_total_text_width = _stream_label.get_theme_font("font").get_string_size(
		full_text,
		HORIZONTAL_ALIGNMENT_LEFT,
		-1,
		_stream_label.get_theme_font_size("font_size")
	).x
	_scroll_offset = 0.0

func _process(delta: float) -> void:
	if _is_paused or _total_text_width <= 0:
		return
	_scroll_offset += scroll_speed * delta
	if _scroll_offset >= _total_text_width:
		_scroll_offset -= _total_text_width
	_stream_label.position.x = -_scroll_offset

2. Left Rail Icon Navigation Button (rail_button.gd)
class_name RailNavButton
extends Button

signal item_selected(route_id: String)

@export var route_id: String = "home"
@export var is_current_route: bool = false:
	set(val):
		is_current_route = val
		queue_redraw()

var is_hovered_state: bool = false

const COLOR_VOID: Color = Color("#0d0c0b")
const COLOR_SURFACE: Color = Color("#1a1816")
const COLOR_HOVER: Color = Color("#22201d")
const COLOR_BORDER: Color = Color("#282521")
const COLOR_GOLD: Color = Color("#f0c233")
const COLOR_INACTIVE: Color = Color("#8c857b")

func _ready() -> void:
	custom_minimum_size = Vector2(56, 48)
	flat = true
	focus_mode = FOCUS_NONE
	mouse_entered.connect(_on_mouse_enter)
	mouse_exited.connect(_on_mouse_exit)
	pressed.connect(func(): item_selected.emit(route_id))

func _on_mouse_enter() -> void:
	is_hovered_state = true
	var tween := create_tween()
	tween.tween_property(self, "modulate:a", 1.0, 0.12).set_trans(Tween.TRANS_QUAD)
	queue_redraw()

func _on_mouse_exit() -> void:
	is_hovered_state = false
	queue_redraw()

func _draw() -> void:
	var r := Rect2(Vector2.ZERO, size)
	
	# Background
	if is_current_route:
		draw_rect(r, COLOR_SURFACE)
	elif is_hovered_state:
		draw_rect(r, COLOR_HOVER)
	
	# Active Gold Left Notch (3px)
	if is_current_route:
		draw_rect(Rect2(0, 8, 3, size.y - 16), COLOR_GOLD)
	
	# Subtle bottom border line
	draw_line(Vector2(8, size.y), Vector2(size.x - 8, size.y), COLOR_BORDER, 1.0)

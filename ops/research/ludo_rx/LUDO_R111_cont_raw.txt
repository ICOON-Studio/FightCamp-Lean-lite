occluding the floor grid and visually obscuring fighters' feet, combat footprints, and interactive station outlines.

In high-end modern commercial boxing gyms and simulation titles like Two Point Museum, interior lighting is driven by overhead diffuse architectural fixtures: suspended continuous linear LED extrusions, high-bay troffers, and recessed flood panels. These fixtures produce short, steep, downward, soft-edged contact shadows rather than raking architectural dusk shadows.

World-Fixed vs. Camera-Rigged Directional Light

Two Point Studios titles decouple the visual key light from a single static world direction when using stepped 90-degree isometric views. If a directional light is purely world-fixed, rotating the camera by 90 degrees can flip the scene from front-quarter lighting into harsh silhouette back-lighting, turning gym equipment into flat dark outlines.

Implementation Rule:

Elevation: Fixed at 74.0° (elevation above horizon). At 74°, shadow projection length ratio is $\cot(74^\circ) \approx 0.287$. A $2.3\text{m}$ pull-up station casts a shadow of just $0.66\text{m}$ directly around its base footprint, acting as a grounded ambient contact shadow rather than a room-spanning barrier.
Azimuth Coupling: The directional light's horizontal yaw slaves to the active camera yaw with a constant local offset of +30.0° (Key). When the player snaps the camera to 45°, 135°, 225°, or 315°, the light rotates synchronously along the ceiling plane, preserving the classic top-left-downward character rim without plunging faces into darkness.

extends Node3D

@export var camera_rig: Node3D
@export var key_light: DirectionalLight3D
@export var world_environment: WorldEnvironment

@export var key_elevation_degrees: float = 74.0
@export var key_azimuth_offset_degrees: float = 30.0
@export var key_energy: float = 2.85
@export var key_color_temperature: float = 5400.0

var _target_yaw: float = 45.0
var _current_yaw: float = 45.0

func _ready() -> void:
	_configure_light()
	_configure_environment()

func _configure_light() -> void:
	if not key_light:
		key_light = DirectionalLight3D.new()
		key_light.name = "CeilingKeyLight"
		add_child(key_light)

	# Godot 4.7 PBR directional setup
	key_light.light_color = Color("#fafafc")
	key_light.light_temperature = key_color_temperature
	key_light.light_energy = key_energy
	key_light.light_indirect_energy = 1.25
	key_light.light_volumetric_fog_energy = 0.0

	key_light.shadow_enabled = true
	key_light.directional_shadow_mode = DirectionalLight3D.SHADOW_PARALLEL_4_SPLITS
	key_light.directional_shadow_split_1 = 0.12
	key_light.directional_shadow_split_2 = 0.28
	key_light.directional_shadow_split_3 = 0.60
	key_light.directional_shadow_blend_splits = true
	key_light.shadow_blur = 2.4
	key_light.shadow_bias = 0.03
	key_light.shadow_normal_bias = 1.8
	key_light.shadow_opacity = 0.72

	_apply_rotation(_current_yaw)

func _configure_environment() -> void:
	if not world_environment:
		return

	var env: Environment = world_environment.environment
	if not env:
		env = Environment.new()
		world_environment.environment = env

	env.background_mode = Environment.BG_COLOR
	env.background_color = Color("#14100e")
	env.ambient_light_source = Environment.AMBIENT_SOURCE_COLOR
	env.ambient_light_color = Color("#e2d8cb")
	env.ambient_light_energy = 0.78
	env.ambient_light_sky_contribution = 0.0

	# Film tone mapping
	env.tonemap_mode = Environment.TONE_MAPPER_FILMIC
	env.tonemap_exposure = 1.15
	env.tonemap_white = 6.0

	# SSAO contact shadowing
	env.ssao_enabled = true
	env.ssao_radius = 0.65
	env.ssao_intensity = 1.8
	env.ssao_power = 1.5
	env.ssao_detail = 0.5
	env.ssao_horizon = 0.06

	# Subtle bloom for high lights
	env.glow_enabled = true
	env.glow_normalized = true
	env.glow_intensity = 0.28
	env.glow_bloom = 0.12
	env.glow_blend_mode = Environment.GLOW_BLEND_MODE_SOFTLIGHT

func on_camera_yaw_changed(new_yaw_deg: float) -> void:
	_target_yaw = new_yaw_deg

func _process(delta: float) -> void:
	if not is_equal_approx(_current_yaw, _target_yaw):
		_current_yaw = rotate_toward(_current_yaw, _target_yaw, delta * 360.0)
		_apply_rotation(_current_yaw)

func _apply_rotation(yaw_deg: float) -> void:
	if not key_light:
		return
	var total_yaw: float = deg_to_rad(yaw_deg + key_azimuth_offset_degrees)
	var pitch: float = deg_to_rad(-key_elevation_degrees)

	var basis := Basis()
	basis = basis.rotated(Vector3.UP, total_yaw)
	basis = basis.rotated(basis.x.normalized(), pitch)
	key_light.global_transform.basis = basis

res://shaders/placement_ghost.gdshader
shader_type spatial;
render_mode blend_mix, depth_draw_always, cull_back;

uniform vec4 ghost_color : source_color = vec4(0.133, 0.773, 0.369, 0.45);
uniform vec4 rim_color : source_color = vec4(1.0, 1.0, 1.0, 0.8);
uniform float rim_power : hint_range(0.5, 8.0) = 3.0;
uniform bool is_invalid = false;

void fragment() {
    vec4 base = ghost_color;
    
    // Invalid hazard stripes calculation
    if (is_invalid) {
        float stripe = sin((SCREEN_UV.x + SCREEN_UV.y) * 120.0);
        if (stripe > 0.3) {
            base.rgb = mix(base.rgb, vec3(0.1, 0.0, 0.0), 0.4);
        }
    }
    
    // Fresnel rim glow for distinct silhouette definition
    float fresnel = pow(1.0 - clamp(dot(NORMAL, VIEW), 0.0, 1.0), rim_power);
    
    ALBEDO = base.rgb;
    ALPHA = clamp(base.a + fresnel * rim_color.a, 0.0, 0.95);
    EMISSION = (base.rgb * 0.4) + (rim_color.rgb * fresnel * 0.6);
}

	_is_valid = is_valid
	if not ghost_material:
		return

		ghost_material.set_shader_parameter("rim_color", Color(0.6, 1.0, 0.6, 0.7))

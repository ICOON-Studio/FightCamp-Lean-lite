class_name MemberCapRead
extends RefCounted

const COLOR_ZERO: Color = Color("#666666")
const COLOR_NORMAL: Color = Color("#E0E0E0")
const COLOR_NEAR: Color = Color("#F5A623")
const COLOR_FULL: Color = Color("#E74C3C")

const LOCKER_TIER_1_PLACES: int = 12
const LOCKER_TIER_2_PLACES: int = 8
const LOCKER_MAX_COUNTED_BANKS: int = 6
const SHOWER_STALL_PLACES: int = 15

static func read(members: int, cap: int, banks: int, showers: int, has_locker_room: bool) -> Dictionary:
	# Hide when system is disabled (-1) or no locker room has been built yet
	if cap < 0 or not has_locker_room:
		return {
			"show": false,
			"text": "-- / --",
			"band": "zero",
			"color": COLOR_ZERO,
			"alert": "",
			"tooltip_lines": [
				"MEMBERSHIP CAPACITY",
				"-- Active Members / -- Maximum Places",
				"Gym closed to public. Complete gym setup to accept members."
			],
			"next_build": "Draw a Locker Room and place a Locker Bank to unlock recurring community membership income."
		}

	var counted_banks: int = clampi(banks, 0, LOCKER_MAX_COUNTED_BANKS)
	var bank_places: int = 0
	for i in range(1, counted_banks + 1):
		if i <= 3:
			bank_places += LOCKER_TIER_1_PLACES
		else:
			bank_places += LOCKER_TIER_2_PLACES
	var shower_places: int = showers * SHOWER_STALL_PLACES

	var band: String = "normal"
	var color: Color = COLOR_NORMAL
	var alert: String = ""
	var next_build: String = ""

	if cap == 0:
		band = "zero"
		color = COLOR_ZERO
		alert = "Nowhere to change - walk-ins turn round"
		next_build = "Next Step: Place 1 Locker Bank ($620) -> Unlocks 12 places."
	elif members >= cap:
		band = "full"
		color = COLOR_FULL
		alert = "The lockers are full"
		if counted_banks < 3:
			next_build = "Next: Place Locker Bank ($620) for +12, or Shower ($1,600) for +15."
		elif counted_banks < LOCKER_MAX_COUNTED_BANKS:
			next_build = "Locker bank scaling dropping to +8 next. Shower Stall ($1,600) gives +15."
		else:
			next_build = "Locker bank limit reached (6/6). Build Showers to expand capacity further."
	elif float(members) / float(cap) >= 0.80:
		band = "near"
		color = COLOR_NEAR
		alert = ""
		if counted_banks < 3:
			next_build = "Next: Place Locker Bank ($620) for +12, or Shower ($1,600) for +15."
		elif counted_banks < LOCKER_MAX_COUNTED_BANKS:
			next_build = "Locker bank scaling dropping to +8 next. Shower Stall ($1,600) gives +15."
		else:
			next_build = "Locker bank limit reached (6/6). Build Showers to expand capacity further."
	else:
		band = "normal"
		color = COLOR_NORMAL
		alert = ""
		if counted_banks < 3:
			next_build = "Next: Place Locker Bank ($620) for +12, or Shower ($1,600) for +15."
		elif counted_banks < LOCKER_MAX_COUNTED_BANKS:
			next_build = "Locker bank scaling dropping to +8 next. Shower Stall ($1,600) gives +15."
		else:
			next_build = "Locker bank limit reached (6/6). Build Showers to expand capacity further."

	var text_display: String = "%d / %d" % [members, cap]
	var tooltip_lines: Array[String] = []

	tooltip_lines.append("MEMBERSHIP CAPACITY")
	tooltip_lines.append("%d Active Members / %d Maximum Places" % [members, cap])

	if cap == 0:
		tooltip_lines.append("No functional locker room detected.")
		tooltip_lines.append("Draw a Locker Room and place a Locker Bank to unlock recurring community membership income.")
		tooltip_lines.append("Status: ALL WALK-INS TURNING AWAY")
	elif band == "full":
		tooltip_lines.append("Locker Banks (%d): +%d places" % [banks, bank_places])
		tooltip_lines.append("Shower Stalls (%d): +%d places" % [showers, shower_places])
		tooltip_lines.append("Status: AT FULL CAPACITY")
		tooltip_lines.append("Walk-ins are currently turning away at reception.")
		tooltip_lines.append(next_build)
	else:
		tooltip_lines.append("Locker Banks (%d): +%d places" % [banks, bank_places])
		tooltip_lines.append("Shower Stalls (%d): +%d places" % [showers, shower_places])
		tooltip_lines.append("Status: ACCEPTING WALK-INS")
		tooltip_lines.append(next_build)

	return {
		"show": true,
		"text": text_display,
		"band": band,
		"color": color,
		"alert": alert,
		"tooltip_lines": tooltip_lines,
		"next_build": next_build
	}

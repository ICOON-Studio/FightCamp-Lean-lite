# scripts/rules/FormBreakdown.gd
# Deterministic rule engine for Form Breakdown & Audibles on the gym floor.
# Pure GDScript 4: RefCounted, static funcs only, zero preloads, no class_name.
extends RefCounted

const COOLDOWN_MINUTES: int = 60
const DAILY_CAP: int = 4
const DECISION_WINDOW_MINUTES: int = 10

const SORENESS_BREAKDOWN_THRESHOLD: float = 65.0

# Base breakdown chance modifiers
const MOD_POWER_THROUGH: float = 1.45
const MOD_PACE_YOURSELF: float = 0.60
const MOD_CRISP_FORM: float = 0.80
const MOD_COACH_PRESENT: float = 0.65

# Station mapping to worked body parts (head_neck, torso_core, left_arm, right_arm, left_leg, right_leg)
static func worked_parts(station_id: String) -> Array:
	match station_id:
		"heavy_bag":
			return ["left_arm", "right_arm", "torso_core"]
		"speed_bag":
			return ["left_arm", "right_arm"]
		"ring_sparring":
			return ["head_neck", "torso_core", "left_arm", "right_arm", "left_leg", "right_leg"]
		"pads":
			return ["left_arm", "right_arm", "torso_core"]
		"defence_drill":
			return ["head_neck", "torso_core", "left_leg", "right_leg"]
		"move_cards":
			return ["torso_core", "left_arm", "right_arm"]
		"treadmill":
			return ["left_leg", "right_leg", "torso_core"]
		"bike":
			return ["left_leg", "right_leg"]
		"recovery":
			return []
		"agility_ladder":
			return ["left_leg", "right_leg"]
		"power_rack":
			return ["left_leg", "right_leg", "torso_core"]
		"olympic_platform":
			return ["left_leg", "right_leg", "torso_core", "right_arm", "left_arm"]
		"weights":
			return ["left_arm", "right_arm", "torso_core"]
		_:
			return ["torso_core"]


static func check(ctx: Dictionary, pick: float) -> Dictionary:
	var minute: int = int(ctx.get("minute", 0))
	var last_change_minute: int = int(ctx.get("last_change_minute", -999))
	var breakdowns_today: int = int(ctx.get("breakdowns_today", 0))
	var form_state: String = String(ctx.get("form_state", "CLEAN"))
	
	if form_state != "CLEAN":
		return {}
	
	if breakdowns_today >= DAILY_CAP:
		return {}
		
	if (minute - last_change_minute) < COOLDOWN_MINUTES:
		return {}

	var station_id: String = String(ctx.get("station_id", ""))
	var parts: Array = worked_parts(station_id)
	if parts.is_empty():
		return {}

	var stamina: float = float(ctx.get("stamina", 100.0))
	var cue: String = String(ctx.get("cue", ""))
	var coach_present: bool = bool(ctx.get("coach_present", false))
	var soreness: Dictionary = Dictionary(ctx.get("soreness", {}))

	# Compute breakdown susceptibility
	var susceptibility: float = 1.0
	if cue == "power_through":
		susceptibility *= MOD_POWER_THROUGH
	elif cue == "pace_yourself":
		susceptibility *= MOD_PACE_YOURSELF
	elif cue == "crisp_form":
		susceptibility *= MOD_CRISP_FORM

	if coach_present:
		susceptibility *= MOD_COACH_PRESENT

	# 1. Soreness Check on primary worked parts
	var worst_part: String = ""
	var max_soreness: float = 0.0
	for p in parts:
		var part_name: String = String(p)
		var val: float = float(soreness.get(part_name, 0.0))
		if val > max_soreness:
			max_soreness = val
			worst_part = part_name

	if max_soreness >= SORENESS_BREAKDOWN_THRESHOLD:
		var soreness_risk: float = ((max_soreness - SORENESS_BREAKDOWN_THRESHOLD) / 35.0) * susceptibility
		if pick < clamp(soreness_risk, 0.15, 0.95):
			return {
				"state": "SLOPPY",
				"part": worst_part,
				"cause": "soreness",
				"gain_mult": 0.50,
				"wear_mult": 1.60,
				"window_minutes": DECISION_WINDOW_MINUTES,
				"toast": "Form Breakdown: %s overloaded!" % worst_part.replace("_", " ").capitalize(),
				"quote": "He's grimacing and favoring the %s. Reps are getting ugly." % worst_part.replace("_", " ")
			}

	# 2. Stamina Band Breakdown Check
	# Bands: FRESH >= 45, WINDED < 45, HEAVY_LEGS < 35, EXHAUSTED < 25
	if stamina < 45.0:
		var stamina_risk: float = 0.0
		if stamina < 25.0:
			stamina_risk = 0.85 * susceptibility
		elif stamina < 35.0:
			stamina_risk = 0.55 * susceptibility
		else:
			stamina_risk = 0.30 * susceptibility

		if pick < clamp(stamina_risk, 0.05, 0.98):
			var active_part: String = String(parts[0]) if parts.size() > 0 else "torso_core"
			return {
				"state": "SLOPPY",
				"part": active_part,
				"cause": "stamina",
				"gain_mult": 0.50,
				"wear_mult": 1.60,
				"window_minutes": DECISION_WINDOW_MINUTES,
				"toast": "Form Breakdown: Exhaustion dragging mechanics down!",
				"quote": "His hips are dragging and his chin is up in the air. Punch mechanics are failing."
			}

	return {}


static func respond(action: String, ctx: Dictionary) -> Dictionary:
	var coach_present: bool = bool(ctx.get("coach_present", false))
	var part: String = String(ctx.get("part", "torso_core"))
	var clean_part_name: String = part.replace("_", " ")

	match action:
		"correct_form":
			var coach_walk_and_drill_time: int = 5 if coach_present else 12
			return {
				"state": "CORRECTED",
				"gain_mult": 1.05,
				"wear_mult": 0.85,
				"coach_minutes": coach_walk_and_drill_time,
				"end_block": false,
				"niggle_chance": 0.0,
				"toast": "Coach Audibles: Form Reset",
				"quote": "Coach steps in, resets the stance, and cues the hips. Clean reps restored."
			}

		"call_rep":
			return {
				"state": "CLEAN",
				"gain_mult": 1.0,
				"wear_mult": 1.0,
				"coach_minutes": 0,
				"end_block": true,
				"niggle_chance": 0.0,
				"toast": "Session Halted Early",
				"quote": "That's enough before bad habits stick. Take off the gloves and head to the mats."
			}

		"ignore":
			return {
				"state": "SLOPPY",
				"gain_mult": 0.30,
				"wear_mult": 2.00,
				"coach_minutes": 0,
				"end_block": false,
				"niggle_chance": 0.25,
				"toast": "Form Ignored: Dirty Reps Accruing!",
				"quote": "Boxer is grinding through severe technical breakdown. Severe injury risk!"
			}

		_:
			return {
				"state": "CLEAN",
				"gain_mult": 1.0,
				"wear_mult": 1.0,
				"coach_minutes": 0,
				"end_block": false,
				"niggle_chance": 0.0,
				"toast": "",
				"quote": ""
			}


static func bubble_color(form_state: String) -> String:
	match form_state:
		"CLEAN", "CORRECTED":
			return "#f0c233" # Clean Gold
		"SLOPPY":
			return "#e6912c" # Warning Amber
		"IGNORED":
			return "#d93829" # Danger Red
		_:
			return "#f0c233"

Design Thresholds & Multipliers Justification
Soreness Trigger Line: Set at >= 70.0 (matching Kalli's K2 Strain that bites). When a body region reaches 70.0 wear, it rolls for a niggle. Base roll is 0.20, scaling up to 0.80 as soreness approaches 100.0.

Requires meaningful time investment: 30-minute standard block or 60-minute intensive treatment.

Push-Through Stakes:

# scripts/rules/Niggle.gd
# Deterministic rule engine for Body Part Niggles, Physio Treatment, and Injury Risk.
# Pure GDScript 4: RefCounted, static funcs only, zero preloads, no class_name.
extends RefCounted

const SORENESS_NIGGLE_LINE: float = 70.0

# Table of body parts loaded by each station (mirrored from FormBreakdown, zero preload)
static func worked_parts(station_id: String) -> Array:
	match station_id:
		"heavy_bag":
			return ["left_arm", "right_arm", "torso_core"]
		"speed_bag":
			return ["left_arm", "right_arm"]
		"ring_sparring":
			return ["head_neck", "torso_core", "left_arm", "right_arm", "left_leg", "right_leg"]
		"pads":
			return ["left_arm", "right_arm", "torso_core"]
		"defence_drill":
			return ["head_neck", "torso_core", "left_leg", "right_leg"]
		"move_cards":
			return ["torso_core", "left_arm", "right_arm"]
		"treadmill":
			return ["left_leg", "right_leg", "torso_core"]
		"bike":
			return ["left_leg", "right_leg"]
		"recovery":
			return []
		"agility_ladder":
			return ["left_leg", "right_leg"]
		"power_rack":
			return ["left_leg", "right_leg", "torso_core"]
		"olympic_platform":
			return ["left_leg", "right_leg", "torso_core", "right_arm", "left_arm"]
		"weights":
			return ["left_arm", "right_arm", "torso_core"]
		_:
			return ["torso_core"]


static func from_soreness(soreness: Dictionary, existing: Array, minute: int, pick: float) -> Dictionary:
	var existing_parts: Array = []
	for n in existing:
		if n is Dictionary:
			existing_parts.append(String(n.get("part", "")))

	var candidates: Array = []
	for part_key in soreness.keys():
		var p: String = String(part_key)
		if existing_parts.has(p):
			continue
		var val: float = float(soreness.get(p, 0.0))
		if val >= SORENESS_NIGGLE_LINE:
			candidates.append({"part": p, "val": val})

	if candidates.is_empty():
		return {}

	# Sort candidates descending by soreness value
	candidates.sort_custom(func(a: Dictionary, b: Dictionary) -> bool:
		return float(a.get("val", 0.0)) > float(b.get("val", 0.0))
	)

	var target: Dictionary = Dictionary(candidates[0])
	var target_part: String = String(target.get("part", ""))
	var target_val: float = float(target.get("val", 70.0))

	# Risk scales from 0.20 at 70 soreness to 0.80 at 100 soreness
	var threshold_span: float = clamp((target_val - SORENESS_NIGGLE_LINE) / 30.0, 0.0, 1.0)
	var chance: float = 0.20 + (threshold_span * 0.60)

	if pick < chance:
		var sev: int = 1
		if target_val >= 90.0 and pick < (chance * 0.5):
			sev = 2

		var part_display: String = target_part.replace("_", " ").capitalize()
		return {
			"part": target_part,
			"severity": sev,
			"minute": minute,
			"toast": "Niggle Warning: %s inflamed!" % part_display,
			"quote": "His %s is visibly locked up. Pushing through risks a major tear." % target_part.replace("_", " ")
		}

	return {}


static func train_effect(niggles: Array, station_id: String) -> Dictionary:
	var parts: Array = worked_parts(station_id)
	if parts.is_empty():
		return {
			"gain_mult": 1.0,
			"wear_mult": 1.0,
			"injury_chance_per_block": 0.0,
			"blocked": false,
			"reason": ""
		}

	var max_sev: int = 0
	var afflicted_part: String = ""

	for n in niggles:
		if not (n is Dictionary):
			continue
		var p: String = String(n.get("part", ""))
		if parts.has(p):
			var s: int = int(n.get("severity", 1))
			if s > max_sev:
				max_sev = s
				afflicted_part = p

	if max_sev == 0:
		return {
			"gain_mult": 1.0,
			"wear_mult": 1.0,
			"injury_chance_per_block": 0.0,
			"blocked": false,
			"reason": ""
		}

	var part_display: String = afflicted_part.replace("_", " ")

	if max_sev >= 3:
		return {
			"gain_mult": 0.10,
			"wear_mult": 2.50,
			"injury_chance_per_block": 0.30,
			"blocked": true,
			"reason": "Station blocked: Severe injury threat to %s!" % part_display
		}
	elif max_sev == 2:
		return {
			"gain_mult": 0.45,
			"wear_mult": 1.80,
			"injury_chance_per_block": 0.12,
			"blocked": false,
			"reason": "High risk: Acute %s strain degrades technique." % part_display
		}
	else:
		return {
			"gain_mult": 0.75,
			"wear_mult": 1.30,
			"injury_chance_per_block": 0.04,
			"blocked": false,
			"reason": "Compromised: Minor %s tweak active." % part_display
		}


static func physio_session(niggle: Dictionary, physio_rating: float, minutes: int) -> Dictionary:
	var sev: int = int(niggle.get("severity", 1))
	var part: String = String(niggle.get("part", "body"))
	var part_display: String = part.replace("_", " ")

	# Effective recovery points: (physio_rating 0-100 / 100) * minutes
	# 30 min @ rating 50 = 15.0 pts.
	# 60 min @ rating 80 = 48.0 pts.
	var work_done: float = (clamp(physio_rating, 10.0, 100.0) / 100.0) * float(minutes)

	var sev_drop: int = 0
	if work_done >= 40.0:
		sev_drop = 2
	elif work_done >= 18.0:
		sev_drop = 1
	elif work_done >= 8.0 and sev == 1:
		sev_drop = 1

	var new_sev: int = max(0, sev - sev_drop)
	var cleared: bool = (new_sev == 0)

	var toast_msg: String = ""
	if cleared:
		toast_msg = "Physio Cleared: %s restored!" % part_display.capitalize()
	else:
		toast_msg = "Physio Managed: %s reduced to Tier %d" % [part_display.capitalize(), new_sev]

	return {
		"cleared": cleared,
		"severity_after": new_sev,
		"minutes_used": minutes,
		"toast": toast_msg
	}


static func push_through(niggle: Dictionary, pick: float) -> Dictionary:
	var sev: int = int(niggle.get("severity", 1))
	var part: String = String(niggle.get("part", "body"))
	var part_display: String = part.replace("_", " ")

	var base_injury_risk: float = 0.08
	if sev == 2:
		base_injury_risk = 0.22
	elif sev >= 3:
		base_injury_risk = 0.50

	if pick < base_injury_risk:
		# Catastrophic structural injury
		var days: int = 7 + int(pick * 21.0) # 7 to 28 days
		return {
			"injured": true,
			"days_out": days,
			"severity_after": 3,
			"toast": "TEAR CONFIRMED: %s ruptured! Out %d days." % [part_display.capitalize(), days]
		}
	else:
		# Escaped immediate rupture, but inflammation worsens
		var next_sev: int = min(3, sev + 1)
		return {
			"injured": false,
			"days_out": 0,
			"severity_after": next_sev,
			"toast": "Pushed Through: %s aggravated to Severity %d." % [part_display.capitalize(), next_sev]
		}


static func day_end(niggles: Array, rested: bool) -> Array:
	var survivors: Array = []
	for n in niggles:
		if not (n is Dictionary):
			continue
		var item: Dictionary = n.duplicate()
		var sev: int = int(item.get("severity", 1))

		if rested:
			sev -= 1
		
		if sev > 0:
			item["severity"] = sev
			survivors.append(item)

	return survivors

	# Test 9: Push through causes catastrophic injury when roll fails
	var push_inj: Dictionary = Niggle.push_through(n_sev2, 0.05) # below 0.22 risk
	if not bool(push_inj.get("injured", false)) or int(push_inj.get("days_out", 0)) < 7:

		failures += 1

	# Test 10: Push through escalates severity when injury avoided
	var push_esc: Dictionary = Niggle.push_through(n_sev1, 0.90) # above 0.08 risk
	if bool(push_esc.get("injured", true)) or int(push_esc.get("severity_after", 0)) != 2:
		print("FAIL: Push-through without injury failed to escalate severity.")
		failures += 1

	# Test 11: day_end healing drops severity by 1 when rested
	var night_niggles: Array = [
		{"part": "left_arm", "severity": 2},
		{"part": "left_leg", "severity": 1}
	]

		print("FAIL: Day-end rest resolution incorrect.")
		failures += 1

	quit(0 if failures == 0 else 1)

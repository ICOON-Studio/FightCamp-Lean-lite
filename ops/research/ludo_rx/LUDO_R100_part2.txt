func set_ledger(status: Dictionary, forecast: Dictionary) -> void:
	var spent: float = float(status.get("spent", 0.0))
	var band: String = String(status.get("band", "FRESH"))
	var line: String = String(status.get("line", ""))

	var projected_spent: float = spent
	if not forecast.is_empty():
		projected_spent = float(forecast.get("projected_spent", spent))

	_target_spent = spent
	_target_ghost = projected_spent

	_text_readout.text = "%.1f / 100" % spent
	_status_line.text = line

	var band_color: Color = COLOR_FRESH
	match band:
		"LOADED":
			band_color = COLOR_LOADED
		"OVER":
			band_color = COLOR_OVER
		_:
			band_color = COLOR_FRESH

	# Detect band change to trigger pulse animation
	if _current_band != "" and _current_band != band:
		_trigger_band_pulse(band_color)
	_current_band = band

	# Calculate bar pixel widths (budget 100 = GAUGE_WIDTH, caps at 120%)
	var spent_ratio: float = clamp(spent / 100.0, 0.0, 1.2)
	var target_spent_px: float = spent_ratio * GAUGE_WIDTH

	var ghost_ratio: float = clamp(projected_spent / 100.0, 0.0, 1.2)
	var target_ghost_px: float = ghost_ratio * GAUGE_WIDTH

	# Smooth tween animations
	if _spent_tween != null and _spent_tween.is_valid():
		_spent_tween.kill()
	_spent_tween = create_tween().set_parallel(true)
	_spent_tween.tween_property(_spent_bar, "size:x", target_spent_px, 0.25).set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_OUT)
	_spent_tween.tween_property(_spent_bar, "color", band_color, 0.25)

	if _ghost_tween != null and _ghost_tween.is_valid():
		_ghost_tween.kill()
	_ghost_tween = create_tween()
	_ghost_tween.tween_property(_ghost_bar, "size:x", target_ghost_px, 0.25).set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_OUT)


func _trigger_band_pulse(pulse_color: Color) -> void:
	if _pulse_tween != null and _pulse_tween.is_valid():
		_pulse_tween.kill()
	_pulse_tween = create_tween()
	_pulse_tween.tween_property(_spent_bar, "color", Color.WHITE, 0.08)
	_pulse_tween.tween_property(_spent_bar, "color", pulse_color, 0.16)

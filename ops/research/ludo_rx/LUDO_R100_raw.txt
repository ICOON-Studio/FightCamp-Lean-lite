static func moments_for(block: Dictionary, boxer: Dictionary, partner: Dictionary, readiness: float) -> Array:
	var minutes: int = int(block.get("minutes", 60))
	var count: int = 1
	if minutes >= 90:
		count = 3
	elif minutes >= 60:
		count = 2

	var triggers: Array = []
	var boxer_stamina: float = float(boxer.get("stamina", 100.0))
	var active_card: String = String(block.get("card", ""))
	var partner_style: String = String(partner.get("style", ""))
	var booked_style: String = String(block.get("booked_opponent_style", ""))

	# Rule 1: Style match has highest priority
	if partner_style != "" and partner_style == booked_style:
		triggers.append("style_match")

	# Rule 2: Active move card being drilled
	if active_card != "":
		triggers.append("move_card")

	# Rule 3: Low stamina introduces fatigue moment
	if boxer_stamina < 40.0:
		triggers.append("fatigue")

	# Rule 4: Stat gap default fill
	while triggers.size() < count:
		triggers.append("stat_gap")

	var planned_moments: Array = []
	# Space moments evenly across the block minutes
	var interval: int = minutes / (count + 1)
	for i in range(count):
		var trig: String = String(triggers[min(i, triggers.size() - 1)])
		var minute_mark: int = interval * (i + 1)
		planned_moments.append({
			"minute": minute_mark,
			"trigger": trig
		})

	return planned_moments


static func moment(trigger: String, ctx: Dictionary, pick: float) -> Dictionary:
	var candidates: Array = []
	for k in MOMENTS.keys():
		var m: Dictionary = Dictionary(MOMENTS[k])
		if String(m.get("trigger", "")) == trigger:
			candidates.append(m)

	if candidates.is_empty():
		candidates.append(Dictionary(MOMENTS["pocket_exchange"]))

	var idx: int = int(floor(pick * float(candidates.size())))
	idx = clamp(idx, 0, candidates.size() - 1)
	var chosen_template: Dictionary = Dictionary(candidates[idx]).duplicate(true)

	var text_variants: Array = Array(chosen_template.get("text_variants", []))
	var chosen_text: String = "Sparring exchange heats up in the center."
	if not text_variants.is_empty():
		var t_idx: int = int(floor(pick * float(text_variants.size())))
		chosen_text = String(text_variants[clamp(t_idx, 0, text_variants.size() - 1)])

	return {
		"id": String(chosen_template.get("id", "pocket_exchange")),
		"title": String(chosen_template.get("title", "Sparring Moment")),
		"text": chosen_text,
		"trigger": trigger,
		"calls": Array(chosen_template.get("calls", []))
	}

static func auto_call(moment_dict: Dictionary, ctx: Dictionary) -> String:
	var calls_list: Array = Array(moment_dict.get("calls", []))
	for c in calls_list:
		if c is Dictionary and bool(c.get("auto", false)):
			return String(c.get("id", ""))

	if not calls_list.is_empty():
		return String(Dictionary(calls_list[0]).get("id", ""))

	return "tie_up"

func _unhandled_input(event: InputEvent) -> void:
	if not visible or _is_resolving:
		return
	if not (event is InputEventKey) or not event.is_pressed() or event.is_echo():
		return

	var key_char: String = OS.get_keycode_string(event.keycode).to_upper()
	for opt in _active_options:
		var k: String = String(opt.get("key", "")).to_upper()
		if k != "" and k == key_char:
			if not bool(opt.get("disabled", false)):
				_on_option_pressed(String(opt.get("id", "")))
				get_viewport().set_input_as_handled()
			return


func _on_option_pressed(opt_id: String) -> void:
	if _is_resolving:
		return

	for opt in _active_options:
		if String(opt.get("id", "")) == opt_id:
			if bool(opt.get("disabled", false)):
				return

	_is_resolving = true
	if _timer_tween != null and _timer_tween.is_valid():
		_timer_tween.kill()

	if _fade_tween != null and _fade_tween.is_valid():
		_fade_tween.kill()
	_fade_tween = create_tween()
	_fade_tween.tween_property(self, "modulate:a", 0.0, 0.120)
	_fade_tween.tween_callback(Callable(self, "_finish_chosen").bind(opt_id))


func _on_timer_timeout() -> void:
	if _is_resolving:
		return
	_on_option_pressed(_auto_option_id)


func _finish_chosen(opt_id: String) -> void:
	hide()
	chosen.emit(opt_id)

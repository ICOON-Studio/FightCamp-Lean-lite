# ==============================================================================
# PlacementValidator.gd
# Pure functional placement validation & walkway reachability engine
# ==============================================================================
class_name PlacementValidator
extends RefCounted

const CELL_SIZE: float = 0.5 # 0.5 meters per tile

# ------------------------------------------------------------------------------
# Footprint & Geometry Helpers
# ------------------------------------------------------------------------------

## Rotates a 2D integer footprint (width, depth in cells) by rotation index (0: 0°, 1: 90°, 2: 180°, 3: 270°)
static func get_rotated_size(w: int, d: int, rot: int) -> Vector2i:
	match rot % 4:
		0, 2:
			return Vector2i(w, d)
		1, 3:
			return Vector2i(d, w)
		_:
			return Vector2i(w, d)

## Computes array of grid cells occupied by an item given its origin, dimensions, and rotation
static func get_occupied_cells(origin: Vector2i, w: int, d: int, rot: int) -> Array[Vector2i]:
	var cells: Array[Vector2i] = []
	var eff_size := get_rotated_size(w, d, rot)
	for x in range(eff_size.x):
		for y in range(eff_size.y):
			cells.append(Vector2i(origin.x + x, origin.y + y))
	return cells

## Computes clearance boundary cells surrounding the solid footprint (front, sides, back)
static func get_clearance_cells(origin: Vector2i, w: int, d: int, rot: int, clearance_pad: int) -> Array[Vector2i]:
	var cells: Array[Vector2i] = []
	var eff_size := get_rotated_size(w, d, rot)
	var min_x := origin.x - clearance_pad
	var max_x := origin.x + eff_size.x + clearance_pad - 1
	var min_y := origin.y - clearance_pad
	var max_y := origin.y + eff_size.y + clearance_pad - 1
	
	for x in range(min_x, max_x + 1):
		for y in range(min_y, max_y + 1):
			var pt := Vector2i(x, y)
			# Only return cells that are outside the solid footprint itself
			var in_solid := (x >= origin.x and x < origin.x + eff_size.x and y >= origin.y and y < origin.y + eff_size.y)
			if not in_solid:
				cells.append(pt)
	return cells

# ------------------------------------------------------------------------------
# Core Validation
# ------------------------------------------------------------------------------

## item: Dictionary:
##   { "id": String, "footprint_w": int, "footprint_d": int, "clearance": int, "wall_mounted": bool }
## cell: Target origin cell (Vector2i)
## rot: Rotation index (0..3)
## occupied_grid: Dictionary of Vector2i -> String (item_id occupying this solid cell)
## hall_bounds: Rect2i (grid bounding rectangle, e.g., 60x36 cells for 30x18 m)
## walls: Array of Dictionaries defining solid wall boundary lines/cells
## entrance_cells: Array[Vector2i] cells representing the exterior door entrance
## all_approach_points: Array[Vector2i] all existing interact/work points that must stay reachable
static func can_place(
	item: Dictionary,
	cell: Vector2i,
	rot: int,
	occupied_grid: Dictionary,
	hall_bounds: Rect2i,
	walls: Array = [],
	entrance_cells: Array = [],
	all_approach_points: Array = []
) -> Dictionary:
	var w: int = item["footprint_w"]
	var d: int = item["footprint_d"]
	var clearance: int = item.get("clearance", 1)
	var wall_mounted: bool = item.get("wall_mounted", false)
	
	var solid_cells := get_occupied_cells(cell, w, d, rot)
	
	# 1. Hall Bounds Check
	for c in solid_cells:
		if not hall_bounds.has_point(c):
			return {"ok": false, "reason": "Out of gym bounds"}
			
	# 2. Solid-on-Solid Overlap Check
	for c in solid_cells:
		if occupied_grid.has(c):
			return {"ok": false, "reason": "Cell already obstructed by %s" % occupied_grid[c]}
			
	# 3. Wall Mounted Rule Check
	if wall_mounted:
		var against_wall := false
		for c in solid_cells:
			# Wall adjacent if sitting on the exterior grid perimeter
			if c.x == hall_bounds.position.x or c.x == hall_bounds.end.x - 1 \
			or c.y == hall_bounds.position.y or c.y == hall_bounds.end.y - 1:
				against_wall = true
				break
		if not against_wall:
			return {"ok": false, "reason": "Item must be mounted flush against a wall"}
			
	# 4. Clearance vs Solid Overlap
	# Clearance zones of the new item cannot overlap existing solid items
	var cl_cells := get_clearance_cells(cell, w, d, rot, clearance)
	for cl in cl_cells:
		if hall_bounds.has_point(cl) and occupied_grid.has(cl):
			return {"ok": false, "reason": "Clearance envelope blocked by %s" % occupied_grid[cl]}

	# 5. Walkway Connectivity BFS Check
	# Temporarily map new solid cells into a test grid to verify all equipment remains pathable
	if not entrance_cells.is_empty():
		var test_grid := occupied_grid.duplicate()
		for c in solid_cells:
			test_grid[c] = item["id"]
			
		# Derive the approach point for this placed item (1 tile in front of the item relative to rotation)
		var this_approach := _get_approach_point(cell, w, d, rot)
		var targets_to_verify: Array[Vector2i] = []
		for pt in all_approach_points:
			targets_to_verify.append(pt)
		targets_to_verify.append(this_approach)
		
		var path_result := verify_walkway_connectivity(entrance_cells[0], targets_to_verify, test_grid, hall_bounds)
		if not path_result.ok:
			return {"ok": false, "reason": "Blocks critical hallway: %s" % path_result.reason}

	return {"ok": true, "reason": "Valid placement"}

static func _get_approach_point(origin: Vector2i, w: int, d: int, rot: int) -> Vector2i:
	var eff_size := get_rotated_size(w, d, rot)
	match rot % 4:
		0: return Vector2i(origin.x + eff_size.x / 2, origin.y + eff_size.y) # Front (+Z / South)
		1: return Vector2i(origin.x - 1, origin.y + eff_size.y / 2)         # West
		2: return Vector2i(origin.x + eff_size.x / 2, origin.y - 1)         # North
		3: return Vector2i(origin.x + eff_size.x, origin.y + eff_size.y / 2) # East
	return origin

# ------------------------------------------------------------------------------
# BFS Navigable Path Checker
# ------------------------------------------------------------------------------

## Runs an unweighted breadth-first flood search across traversable grid tiles.
## Guarantees an open path from the gym entrance to every station's work/approach node.
static func verify_walkway_connectivity(
	start_cell: Vector2i,
	required_destinations: Array,
	blocked_cells: Dictionary,
	bounds: Rect2i
) -> Dictionary:
	if blocked_cells.has(start_cell):
		return {"ok": false, "reason": "Entrance blocked"}
		
	var queue: Array[Vector2i] = [start_cell]
	var visited: Dictionary = {start_cell: true}
	var reachable_count: int = 0
	
	var dest_map: Dictionary = {}
	for d in required_destinations:
		dest_map[d] = false
		if d == start_cell:
			dest_map[d] = true
			reachable_count += 1
			
	var directions: Array[Vector2i] = [
		Vector2i(1, 0), Vector2i(-1, 0), Vector2i(0, 1), Vector2i(0, -1)
	]
	
	var head := 0
	while head < queue.size():
		var current = queue[head]
		head += 1
		
		for dir in directions:
			var neighbor: Vector2i = current + dir
			
			if not bounds.has_point(neighbor):
				continue
			if visited.has(neighbor):
				continue
			if blocked_cells.has(neighbor):
				continue
				
			visited[neighbor] = true
			queue.append(neighbor)
			
			if dest_map.has(neighbor) and dest_map[neighbor] == false:
				dest_map[neighbor] = true
				reachable_count += 1
				if reachable_count >= dest_map.size():
					return {"ok": true, "reason": "All stations reachable"}

	for dest in dest_map.keys():
		if not dest_map[dest]:
			return {"ok": false, "reason": "Station at (%d, %d) is isolated" % [dest.x, dest.y]}
			
	return {"ok": true, "reason": "All stations reachable"}

C1. Analytical Comparison: Where Our Numbers Differ and Why

The telemetry extracted from Two Point Museum (TPM, 2025) provides an exceptional empirical baseline from the genre’s leading production team. Here is how our engineering choices compare, along with tailored adaptations for FightCamp:

CAMERA SYSTEM COMPARISON:
Two Point Museum: [FOV 17° (Near-Ortho)] ──> [Height 20-280m] ──> [Free Yaw Drag @ 90°/s]
FightCamp Spec:   [FOV 24° (Sports Depth)] ──> [Height 12-45m]  ──> [Discrete 90° Snapping]

1. Camera Projection & FOV: 17° vs. 36°
Measured TPM: Perspective with FOV $17^\circ$ (effectively an orthographic telephoto perspective, completely suppressing perspective distortion across vast $100\text{ m}+$ museum complexes).
FightCamp Original: FOV $36^\circ$ to $48^\circ$.
The Conflict: At FOV $36^\circ$ to $48^\circ$, tall characters ($1.8\text{ m}$) standing near the near camera edge exhibit intense perspective barrel elongation, distorting their proportions. However, at $17^\circ$, depth cues flatten so heavily that ring ropes, apron steps, and fighter bob-and-weave verticality lose their physical punch.
Reconciliation: Adopt FOV $24^\circ$. This sits directly in the sweet spot: it delivers the near-orthographic readability of a technical management sim while retaining sufficient foreshortening to showcase muscular boxing animations and ring ropes.
2. Camera Rotation: Free Yaw (90°/s) vs. Discrete 90-Degree Steps
Measured TPM: Free continuous yaw rotation via keys ($90^\circ/\text{s}$) or mouse drag, backed by continuous shader updates (SetWallShaderToTransparentMode).
FightCamp PART A: Fixed discrete $90^\circ$ steps ($45^\circ \to 135^\circ \to 225^\circ \to 315^\circ$).
The Conflict: Continuous free rotation in Godot 4 with dynamic cutaway geometry requires either per-frame backface cull passes or runtime dissolve shaders across materials, increasing shader complexity. More importantly, FightCamp features active real-time athletic interactions (coach boosting a fighter at a bag, pad chains executing in the ring) occurring on strict 2-minute clocks. Searching for a manual rotation angle during active gameplay is frustrating.
Reconciliation: Retain Discrete $90^\circ$ Steps as the Default Game Control (Q/E), but adopt TPM’s Free Drag ($90^\circ/\text{s}$) strictly during Build Mode (B). When constructing or arranging equipment, the player gains full rotational freedom to inspect alignments. The moment they return to simulation mode, the camera snaps smoothly to the nearest of the four standard isometric quadrants ($45^\circ, 135^\circ, 225^\circ, 315^\circ$).
3. Wall Rendering: Backface / Shader Cut vs. Geometric Stub
Measured TPM: ShowBackfaceWallsOnly on far walls, and SetWallShaderToTransparentMode on camera-facing walls. Visual wall height is completely decoupled from selection/placement wall height. Ceiling height is $6\text{ m}$.
FightCamp Part A: Geometric physical drop to a $0.4\text{ m}$ stub with cap rails.
Reconciliation: Adopt TPM’s Decoupling Principle with FightCamp’s Stub Geometry.
Visuals: Keep our $0.4\text{ m}$ physical stub cap rail. Transparent walls look washed out over brick gym aesthetics. A clean architectural cutaway is visually clearer than a transparent dither.
Selection & Snapping: Adopt TPM’s exact decoupled logic: even when a wall is cut down to a $0.4\text{ m}$ visual stub, its virtual placement plane remains at full $3.6\text{ m}$ height. When the player drags a poster or round clock near the stub, the full wall highlight appears, snapping the item cleanly to its correct height ($1.8\text{ m}$).
4. Grid Snap: 0.5 m Grid Density (87% Dominance in TPM)
Measured TPM: $0.5\text{ m}$ on 87% of items, $1.0\text{ m}$ on 5%, fine $0.25/0.1\text{ m}$ on decor. Free (off-grid) placement allowed on 99% of items.
FightCamp Part B: Proposed $0.5\text{ m}$ uniform grid.
Reconciliation: Full Alignment. TPM proves conclusively that $0.5\text{ m}$ is the operational standard for top-down sim management. We adopt $0.5\text{ m}$ for all training stations, $1.0\text{ m}$ for the boxing ring platform footprint, and allow free off-grid placement (Alt hold) strictly for small decorative floor objects (chalk buckets, water bottles, gym bags).
5. Room Sizing & Prestige Mechanics
Measured TPM: Minimum room sizes (Bathrooms $2 \times 1\text{ cells}$, Staff $3 \times 2$, Workshops $4 \times 4$); MaxPrestigePerCell clamped between 30 and 48; construction grows dynamically ($20% \to 95%$ over 2 seconds).
FightCamp Adaptation: Our support rooms match TPM's exact scale curves: Physio ($3 \times 3\text{ m} \to 6 \times 6\text{ cells}$), Locker Room ($4 \times 3\text{ m} \to 8 \times 6\text{ cells}$). We adopt TPM’s MaxPrestigePerCell architecture to prevent exploit exploits (e.g., players cramming 50 boxing trophies into a $2 \times 2\text{ m}$ closet to artificially inflate gym rating).

Complete Selection Algorithm & Ranking Pipeline (GDScript / C# Architecture)

Every time the bell signals the end of a round, the following algorithm executes inside CornerManager.cs:

public struct CoachRecommendation
{
    public string Channel;         // "OPPONENT", "RISK", "ECONOMY"
    public string DialogueText;
    public string ButtonLabel;
    public TacticalOrder OrderData;
    public float PriorityScore;
}

public List<CoachRecommendation> EvaluateRound(RoundAuditData round, Fighter player, Fighter opponent)
{
    List<CoachRecommendation> recs = new List<CoachRecommendation>();

    // SLOT 1: OPPONENT VULNERABILITIES (Score: 0 to 100)
    var slot1Candidates = new List<CoachRecommendation>();
    
    // Check 1.1: Opponent Whiffs
    float oppWhiffRate = (float)opponent.PunchesWhiffedRound / Mathf.Max(1, opponent.PunchesThrownRound);
    if (oppWhiffRate >= 0.40f && opponent.PunchesThrownRound >= 10)
    {
        slot1Candidates.Add(new CoachRecommendation {
            PriorityScore = 80f + (oppWhiffRate * 20f),
            DialogueText = $"His {opponent.MostWhiffedPunchType} is finding pure air — slipped {opponent.PunchesWhiffedRound} of {opponent.PunchesThrownRound}. Make him pay on the miss.",
            ButtonLabel = "EXPLOIT WHIFF COUNTERS",
            OrderData = TacticalOrder.ExploitCounters
        });
    }
    // Check 1.2: Opponent Chin
    if (opponent.ChinShockOccurredThisRound || opponent.EffectiveChin <= 45)
    {
        slot1Candidates.Add(new CoachRecommendation {
            PriorityScore = 75f + (100 - opponent.EffectiveChin) * 0.25f,
            DialogueText = $"His legs went soft on the clean right. His chin rating ({opponent.EffectiveChin:F0}) won't survive another flush exchange—sit down on the right hand.",
            ButtonLabel = "LOAD UP REAR POWER",
            OrderData = TacticalOrder.LoadUpRearPower
        });
    }
    // Check 1.3: Low Accuracy
    float oppAcc = (float)opponent.PunchesLandedRound / Mathf.Max(1, opponent.PunchesThrownRound);
    if (oppAcc <= 0.22f && opponent.PunchesThrownRound >= 12)
    {
        slot1Candidates.Add(new CoachRecommendation {
            PriorityScore = 65f + (0.22f - oppAcc) * 100f,
            DialogueText = $"He is lunging blind ({(oppAcc * 100f):F0}% this round). Let him initiate, hold your pocket, and rip immediately after the second beat.",
            ButtonLabel = "FIGHT OFF BACK FOOT",
            OrderData = TacticalOrder.BackFootCounters
        });
    }
    // Default Slot 1 Fallback
    slot1Candidates.Add(new CoachRecommendation {
        PriorityScore = 1.0f,
        DialogueText = "He's resetting to his left every time he finishes. Circle right to take away his power side.",
        ButtonLabel = "CIRCLE TO POWER SIDE",
        OrderData = TacticalOrder.CirclePowerSide
    });
    slot1Candidates.Sort((a, b) => b.PriorityScore.CompareTo(a.PriorityScore));
    recs.Add(slot1Candidates[0]);

    // SLOT 2: PLAYER STRUCTURAL RISK (Score: 0 to 100)
    var slot2Candidates = new List<CoachRecommendation>();
    
    // Check 2.1: Chin Danger / Head Damage
    if (player.HeadDamageTakenThisRound >= 8.0f || player.ChinShockOccurredThisRound)
    {
        slot2Candidates.Add(new CoachRecommendation {
            PriorityScore = 90f + player.HeadDamageTakenThisRound,
            DialogueText = $"Your head snapped back twice off his {opponent.MostLandedPunchType}. Tuck that chin behind the shoulder and pick it off with the gloves.",
            ButtonLabel = "HIGH SHELL GUARD",
            OrderData = TacticalOrder.HighShellGuard
        });
    }
    // Check 2.2: Specific Punch Leaking
    var worstLeaked = player.GetMostLeakedPunchAgainst();
    if (worstLeaked.LandedAgainst >= 6 && worstLeaked.BlockRate < 0.20f)
    {
        slot2Candidates.Add(new CoachRecommendation {
            PriorityScore = 70f + worstLeaked.LandedAgainst * 2f,
            DialogueText = $"His {worstLeaked.PunchName} is finding your jaw every exchange ({worstLeaked.LandedAgainst}/{worstLeaked.ThrownAgainst} landed). Keep that opposite glove pinned to your ear.",
            ButtonLabel = $"ADAPT GUARD: {worstLeaked.PunchName.ToUpper()}",
            OrderData = TacticalOrder.AdaptGuard(worstLeaked.PunchType)
        });
    }
    // Check 2.3: Footwork / Leg Wear
    if (player.LegsCondition <= 80f)
    {
        slot2Candidates.Add(new CoachRecommendation {
            PriorityScore = 60f + (80f - player.LegsCondition),
            DialogueText = $"Your feet are flat—evasion fell to {player.EffectiveEvasion:F0}. Stop bouncing in place; plant yourself and make him come through the guard.",
            ButtonLabel = "PLANT & HOLD CENTER",
            OrderData = TacticalOrder.PlantAndHold
        });
    }
    // Default Slot 2 Fallback
    slot2Candidates.Add(new CoachRecommendation {
        PriorityScore = 1.0f,
        DialogueText = $"Defensively clean that round—only {player.CleanPunchesAbsorbedRound} clean shots slipped through. Keep the hands active and don't get complacent.",
        ButtonLabel = "MAINTAIN CURRENT GUARD",
        OrderData = TacticalOrder.MaintainGuard
    });
    slot2Candidates.Sort((a, b) => b.PriorityScore.CompareTo(a.PriorityScore));
    recs.Add(slot2Candidates[0]);

    // SLOT 3: ENERGY & ATTRITION (Score: 0 to 100)
    var slot3Candidates = new List<CoachRecommendation>();
    
    // Check 3.1: Whiff Burn
    if (player.StaminaLostToWhiffsRound >= 6.0f)
    {
        slot3Candidates.Add(new CoachRecommendation {
            PriorityScore = 85f + (player.StaminaLostToWhiffsRound * 2f),
            DialogueText = $"You're bleeding wind throwing at shadows—{player.PunchesWhiffedRound} misses burned {player.StaminaLostToWhiffsRound:F1} stamina units. Stop loading up from the bleachers.",
            ButtonLabel = "PUNCH ON CONFIRMATION",
            OrderData = TacticalOrder.PunchOnConfirmation
        });
    }
    // Check 3.2: Gassed
    if (player.CurrentStamina <= 30.0f)
    {
        slot3Candidates.Add(new CoachRecommendation {
            PriorityScore = 88f + (30f - player.CurrentStamina),
            DialogueText = $"You're running on fumes ({player.CurrentStamina:F0}% tank). Give away the first minute of this next round: tie him up, breathe, and survive.",
            ButtonLabel = "CLINCH & REST ROUND",
            OrderData = TacticalOrder.ClinchAndRest
        });
    }
    // Default Slot 3 Fallback
    slot3Candidates.Add(new CoachRecommendation {
        PriorityScore = 1.0f,
        DialogueText = $"Pacing looks solid ({player.CurrentStamina:F0}% remaining). Maintain this rhythm and don't get sucked into an ugly brawl.",
        ButtonLabel = "PRESERVE TEMPO",
        OrderData = TacticalOrder.PreserveTempo
    });
    slot3Candidates.Sort((a, b) => b.PriorityScore.CompareTo(a.PriorityScore));
    recs.Add(slot3Candidates[0]);

    return recs; // Exactly 3 targeted directives
}

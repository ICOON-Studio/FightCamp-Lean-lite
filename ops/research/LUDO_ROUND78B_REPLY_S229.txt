FIGHTCAMP DESIGN DIRECTIVE: S229 / ROUND 78
MARKET SELECTION CARD REDESIGN ("CHOOSE THE MARKET")
PART A: DEFECT AUDIT OF CURRENT SHIPPING CARD

Looking at the current United Kingdom market panel crop (480 px fixed panel):

Extreme Text-Heavy Barrenness (Zero Visual Identity): The card looks like an unstyled developer debug inspector. Aside from an isolated, low-resolution 2D country flag thumbnail and a single brass crest glyph, there is zero environmental or boxing world art. A player choosing a starting home turf wants grit, atmosphere, and prestige (the mist over Bethnal Green, the York Hall brick, the neon glare of an arena).
Hairline, Anemic Pillar Bars: The current progress bars are flat, thin 4-pixel strips with mismatched, harsh colors (a jarring primary yellow, pure saturated red, and dull mustard) that ignore the Round-70 visual token hierarchy. They lack physical weight, tick divisions, depth beveling, or tactile tactile impact.
Redundant & Low-Value Data Clutter ("Active Hub Dossier"): The card devotes massive vertical space to reading out generic local gym names ("London Boxing", "Vanguard Training Centre") in plain white text. At the market selection stage, the player does not care about individual rival AI gym names; they care about macro competition, national talent level, economic viability, and who reigns at the top of the food chain.
Governing Bodies Dead-End: "BRITISH BOXING BOARD OF CONTROL / +1 MORE" sits inside a boxed panel that gives zero gameplay mechanics or actionable friction/rules to the player. It operates as dead typographic weight.
Absent Competitive Stakes (No Star Power): Boxing is an athlete-driven star sport. A country select screen that does not showcase the national champions or top pound-for-pound prospects fails to trigger the player's core fantasy: "Can my gym produce someone who can dethrone the UK's #1 heavyweight or outbox their golden boy?"
Muted Typography & Flat Hierarchy: The letter grades ("GRADE S", "GRADE A+", "GRADE B+") share identical font size, weight, and tracking as the numerical scores ("92 / 100"). The tier tag ("TIER 1 COMBAT MARKET") is trapped inside an awkward bordered box with excessive empty space.
Missing Primary Action Anchor: The crop lacks the fixed, punchy CTA button ("ESTABLISH GYM IN THIS MARKET" or "CHOOSE MARKET") styled in #f0c233 Championship Gold.

PART E: GDSCRIPT (GODOT 4.7) IMPLEMENTATION
1. SegmentedBar.gd (Thick Notched Pillar Bar)
# SegmentedBar.gd
# Round-70 compliant thick segmented meter with tween animations.
@tool
class_name SegmentedBar
extends Control

signal animation_completed

@export var max_value: float = 100.0
@export var current_value: float = 0.0:
	set(val):
		current_value = clamp(val, 0.0, max_value)
		queue_redraw()

@export var segments: int = 10
@export var bar_height: float = 14.0
@export var segment_gap: float = 2.0
@export var corner_radius: float = 2.0

# Round-70 Color Tokens
const COLOR_VOID: Color = Color("#0d0c0b")
const COLOR_TROUGH: Color = Color("#141210")
const COLOR_BORDER: Color = Color("#282521")
const COLOR_GOLD_HIGH: Color = Color("#f0c233")
const COLOR_GOLD_LOW: Color = Color("#c87d2a")
const COLOR_MUTED_BAR: Color = Color("#78726a")

var _display_value: float = 0.0
var _tween: Tween

func _ready() -> void:
	custom_minimum_size.y = bar_height

func animate_to_value(target_val: float, duration: float = 0.45, delay: float = 0.0) -> void:
	if _tween and _tween.is_running():
		_tween.kill()
	
	_tween = create_tween().set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_OUT)
	if delay > 0.0:
		_tween.tween_interval(delay)
	
	_tween.tween_property(self, "_display_value", clamp(target_val, 0.0, max_value), duration)
	_tween.connect("step_finished", Callable(self, "_on_tween_step"))
	_tween.connect("finished", Callable(self, "_on_tween_finished"))

func _on_tween_step(_idx: int) -> void:
	queue_redraw()

func _on_tween_finished() -> void:
	current_value = _display_value
	emit_signal("animation_completed")

func _draw() -> void:
	var total_w: float = size.x
	var available_w: float = total_w - (segment_gap * (segments - 1))
	var seg_w: float = available_w / float(segments)
	
	var active_ratio: float = _display_value / max_value
	var filled_segments_float: float = active_ratio * float(segments)
	
	# Determine color grading based on value
	var fill_color: Color = COLOR_GOLD_HIGH
	if _display_value < 70.0:
		fill_color = COLOR_MUTED_BAR
	elif _display_value < 85.0:
		fill_color = COLOR_GOLD_LOW
		
	for i in range(segments):
		var seg_x: float = i * (seg_w + segment_gap)
		var seg_rect: Rect2 = Rect2(seg_x, (size.y - bar_height) * 0.5, seg_w, bar_height)
		
		# Draw Background Trough Segment
		draw_rect(seg_rect, COLOR_TROUGH, true)
		draw_rect(seg_rect, COLOR_BORDER, false, 1.0)
		
		# Draw Active Fill Segment
		if float(i + 1) <= filled_segments_float:
			# Fully filled segment
			draw_rect(seg_rect.grow(-1.0), fill_color, true)
		elif float(i) < filled_segments_float:
			# Partially filled segment
			var partial_ratio: float = filled_segments_float - float(i)
			var partial_w: float = (seg_w - 2.0) * partial_ratio
			var partial_rect: Rect2 = Rect2(seg_rect.position + Vector2(1.0, 1.0), Vector2(partial_w, bar_height - 2.0))
			draw_rect(partial_rect, fill_color, true)

2. TopFighterRow.gd (Top 5 Fighter List Item)
# TopFighterRow.gd
# Round-70 compliant fighter item row displaying national star profile.
class_name TopFighterRow
extends PanelContainer

signal fighter_inspected(fighter_id: String)

@onready var rank_label: Label = %RankLabel
@onready var portrait_rect: TextureRect = %PortraitRect
@onready var name_label: Label = %NameLabel
@onready var record_label: Label = %RecordLabel
@onready var division_label: Label = %DivisionLabel
@onready var status_badge: Label = %StatusBadge

# Round-70 Color & Style Constants
const COLOR_BASE_PANEL: Color = Color("#1a1816")
const COLOR_HOVER_PANEL: Color = Color("#22201d")
const COLOR_BORDER_DEFAULT: Color = Color("#282521")
const COLOR_BORDER_HOVER: Color = Color("#4a4338")
const COLOR_GOLD: Color = Color("#f0c233")
const COLOR_CREAM: Color = Color("#e6e1d8")
const COLOR_ASH: Color = Color("#8c857b")

var _fighter_data: Dictionary = {}
var _stylebox: StyleBoxFlat

func _ready() -> void:
	custom_minimum_size.y = 48.0
	mouse_filter = Control.MOUSE_FILTER_STOP
	
	_stylebox = StyleBoxFlat.new()
	_stylebox.bg_color = COLOR_BASE_PANEL
	_stylebox.border_color = COLOR_BORDER_DEFAULT
	_stylebox.set_border_width_all(1)
	_stylebox.set_corner_radius_all(4)
	add_theme_stylebox_override("panel", _stylebox)
	
	mouse_entered.connect(_on_mouse_entered)
	mouse_exited.connect(_on_mouse_exited)

func populate(rank: int, data: Dictionary) -> void:
	_fighter_data = data
	rank_label.text = "#%d" % rank
	if rank == 1:
		rank_label.add_theme_color_override("font_color", COLOR_GOLD)
	else:
		rank_label.add_theme_color_override("font_color", COLOR_ASH)
		
	name_label.text = data.get("name", "Unknown Fighter")
	var wins: int = data.get("wins", 0)
	var losses: int = data.get("losses", 0)
	var draws: int = data.get("draws", 0)
	var kos: int = data.get("kos", 0)
	record_label.text = "%d-%d-%d (%d KO)" % [wins, losses, draws, kos]
	division_label.text = data.get("weight_class", "Middleweight").to_upper()
	
	var status: String = data.get("status", "")
	if status != "":
		status_badge.visible = true
		status_badge.text = status.to_upper()
	else:
		status_badge.visible = false
		
	if data.has("portrait_texture"):
		portrait_rect.texture = data["portrait_texture"]

func _gui_input(event: InputEvent) -> void:
	if event is InputEventMouseButton and event.button_index == MOUSE_BUTTON_LEFT and event.pressed:
		accept_event()
		emit_signal("fighter_inspected", _fighter_data.get("id", ""))

func _on_mouse_entered() -> void:
	var tween: Tween = create_tween().set_parallel(true).set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_OUT)
	tween.tween_property(_stylebox, "bg_color", COLOR_HOVER_PANEL, 0.18)
	tween.tween_property(_stylebox, "border_color", COLOR_BORDER_HOVER, 0.18)
	name_label.add_theme_color_override("font_color", COLOR_GOLD)

func _on_mouse_exited() -> void:
	var tween: Tween = create_tween().set_parallel(true).set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_OUT)
	tween.tween_property(_stylebox, "bg_color", COLOR_BASE_PANEL, 0.18)
	tween.tween_property(_stylebox, "border_color", COLOR_BORDER_DEFAULT, 0.18)
	name_label.add_theme_color_override("font_color", COLOR_CREAM)

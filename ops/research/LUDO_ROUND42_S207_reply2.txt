B6. How It Must NOT Feel (Anti-Frustration & Sensitivity Mandates)
Zero Unexplained RNG Spikes (Readable Causality):
A fighter must never break without an unambiguous paper trail.
Every point of stress gained or lost is logged into fighter.stress_log with an explicit causative string (e.g., [WEEK 4, DAY 2] +5 Stress: Sustained heavy knockdown in sparring against D. Cole).
When the resolve check triggers at 100, the resulting affliction is not a random dice roll across the entire database; it is tied directly to the dominant stress category over the past 28 days.
No Arbitrary Stat Decay (Punch Club Anti-Pattern):
Stress and afflictions never decay permanent baseline stats (power, technique, speed, defense, conditioning, fight_iq).
Afflictions act as dynamic multipliers and behavioral overrides that throttle current training efficiency or alter fight-night tactical behavior. Once cured, the fighter returns to their full athletic capability.
No Clinical Psychiatric Depictions (Mental Health Sensitivity):
The UI strictly forbids clinical psychiatric terminology: no "Depressed", "Manic", "Neurotic", or "Bipolar" labels.
Terminology is anchored purely in professional boxing vernacular: OVERCOOKED (overtraining syndrome/NFOR), GUN-SHY (post-concussion/hesitation syndrome), STARVING (acute dehydration/glycogen depletion), FRAGILE (structural tissue overload), STEELED (flow state/fortitude).
No depictions of substance abuse, self-harm, or existential breakdown. The struggle is purely that of a dedicated combat athlete managing physical exhaustion, ring fear, and the physical toll of weight management.

# StressAfflictionEngine.gd
# Pure functional calculation module for stress, afflictions, and virtues.
# Zero scene tree dependencies. Operates strictly on plain dictionaries and typed primitives.
class_name StressAfflictionEngine

enum AfflictionType {
    NONE,
    OVERCOOKED,
    GUN_SHY,
    STARVING,
    FRAGILE
}

enum VirtueType {
    NONE,
    STEELED
}

# Evaluates composure modifier on raw stress gains
static func calculate_effective_stress_gain(base_gain: float, composure: int) -> float:
    var clamped_composure: float = clampf(float(composure), 0.0, 100.0)
    var multiplier: float = 1.6 - (clamped_composure / 100.0)
    return maxf(0.0, base_gain * multiplier)

# Appends stress event, caps at 100, tracks dominant category
static func apply_stress_event(fighter: Dictionary, base_gain: float, category: String, reason: String) -> Dictionary:
    var effective_gain: float = calculate_effective_stress_gain(base_gain, fighter.get("composure", 50))
    fighter["stress"] = clampf(fighter.get("stress", 0.0) + effective_gain, 0.0, 100.0)
    
    # Track category history for deterministic resolve check
    if not fighter.has("stress_history"):
        fighter["stress_history"] = {"training": 0.0, "trauma": 0.0, "weight_cut": 0.0, "wear": 0.0}
    fighter["stress_history"][category] = fighter["stress_history"].get(category, 0.0) + effective_gain
    
    # Append to human-readable log
    if not fighter.has("stress_log"):
        fighter["stress_log"] = []
    fighter["stress_log"].push_front({
        "delta": effective_gain,
        "reason": reason,
        "total": fighter["stress"]
    })
    if fighter["stress_log"].size() > 20:
        fighter["stress_log"].pop_back()
        
    return fighter

# Executes the resolve check when stress hits 100
static func evaluate_resolve_check(fighter: Dictionary, deterministic_roll: float = -1.0) -> Dictionary:
    if fighter.get("stress", 0.0) < 100.0:
        return fighter # Threshold not reached
        
    var composure: int = fighter.get("composure", 50)
    var virtue_chance: float = 15.0 + (float(composure) / 5.0) # 15% to 35%
    var roll: float = deterministic_roll if deterministic_roll >= 0.0 else randf_range(0.0, 100.0)
    
    if roll <= virtue_chance:
        fighter["active_virtue"] = VirtueType.STEELED
        fighter["active_affliction"] = AfflictionType.NONE
        fighter["virtue_weeks_remaining"] = 4
        fighter["stress"] = 50.0 # Relief reset
    else:
        fighter["active_virtue"] = VirtueType.NONE
        fighter["virtue_weeks_remaining"] = 0
        fighter["stress"] = 60.0 # Catharsis reset
        
        # Pick dominant stress source
        var history: Dictionary = fighter.get("stress_history", {})
        var dominant_cat: String = "training"
        var max_val: float = -1.0
        for cat in history.keys():
            if history[cat] > max_val:
                max_val = history[cat]
                dominant_cat = cat
                
        match dominant_cat:
            "training":
                fighter["active_affliction"] = AfflictionType.OVERCOOKED
                fighter["affliction_duration_weeks"] = 4
            "trauma":
                fighter["active_affliction"] = AfflictionType.GUN_SHY
                fighter["affliction_duration_weeks"] = 6
            "weight_cut":
                fighter["active_affliction"] = AfflictionType.STARVING
                fighter["affliction_duration_weeks"] = 2
            "wear":
                fighter["active_affliction"] = AfflictionType.FRAGILE
                fighter["affliction_duration_weeks"] = 5
            _:
                fighter["active_affliction"] = AfflictionType.OVERCOOKED
                fighter["affliction_duration_weeks"] = 4
                
    return fighter

# Modifies in-fight attributes based on active mental state
static func apply_fight_night_stat_modifiers(fighter: Dictionary, raw_stats: Dictionary) -> Dictionary:
    var modded_stats: Dictionary = raw_stats.duplicate()
    var affliction = fighter.get("active_affliction", AfflictionType.NONE)
    var virtue = fighter.get("active_virtue", VirtueType.NONE)
    
    if virtue == VirtueType.STEELED:
        modded_stats["composure"] = 99
        return modded_stats
        
    match affliction:
        AfflictionType.OVERCOOKED:
            # Stamina decay accelerated, defense eroded in late rounds
            modded_stats["conditioning"] = max(1, int(modded_stats.get("conditioning", 50) * 0.85))
        AfflictionType.GUN_SHY:
            # Reluctant to commit power shots
            modded_stats["power"] = max(1, modded_stats.get("power", 50) - 10)
            modded_stats["fight_iq"] = max(1, modded_stats.get("fight_iq", 50) - 5)
        AfflictionType.STARVING:
            # Drained by cut: baseline conditioning drops, chin drops
            modded_stats["conditioning"] = max(1, modded_stats.get("conditioning", 50) - 15)
            modded_stats["defense"] = max(1, modded_stats.get("defense", 50) - 8)
        AfflictionType.FRAGILE:
            # Protecting an injured zone compromises general defense
            modded_stats["defense"] = max(1, modded_stats.get("defense", 50) - 12)
            
    return modded_stats

# test_stress_afflictions.gd
extends GutTest

func test_composure_correctly_scales_stress_accumulation():
    var high_comp_gain = StressAfflictionEngine.calculate_effective_stress_gain(10.0, 90) # 1.6 - 0.9 = 0.7x
    var low_comp_gain = StressAfflictionEngine.calculate_effective_stress_gain(10.0, 20)  # 1.6 - 0.2 = 1.4x
    
    assert_almost_eq(high_comp_gain, 7.0, 0.01, "90 Composure must reduce stress gain to 70%")
    assert_almost_eq(low_comp_gain, 14.0, 0.01, "20 Composure must amplify stress gain to 140%")

func test_stress_caps_at_100_and_does_not_overflow():
    var fighter = {"composure": 50, "stress": 95.0}
    fighter = StressAfflictionEngine.apply_stress_event(fighter, 20.0, "training", "Extreme sparring")
    assert_eq(fighter["stress"], 100.0, "Stress meter must clamp at 100")

func test_resolve_check_triggers_steeled_virtue_on_success():
    var fighter = {"composure": 50, "stress": 100.0} # 15 + (50/5) = 25% virtue chance
    # Deterministic roll of 15.0% <= 25.0% -> Virtue fires
    fighter = StressAfflictionEngine.evaluate_resolve_check(fighter, 15.0)
    
    assert_eq(fighter["active_virtue"], StressAfflictionEngine.VirtueType.STEELED)
    assert_eq(fighter["active_affliction"], StressAfflictionEngine.AfflictionType.NONE)
    assert_eq(fighter["stress"], 50.0, "Stress must reset to 50 on virtue")

func test_resolve_check_picks_dominant_stress_affliction():
    var fighter = {
        "composure": 20, # 19% virtue chance
        "stress": 100.0,
        "stress_history": {
            "training": 10.0,
            "trauma": 45.0, # DOMINANT (e.g. KO loss + knockdown)
            "weight_cut": 0.0,
            "wear": 5.0
        }
    }
    # Deterministic roll of 50.0% > 19% -> Affliction fires
    fighter = StressAfflictionEngine.evaluate_resolve_check(fighter, 50.0)
    
    assert_eq(fighter["active_virtue"], StressAfflictionEngine.VirtueType.NONE)
    assert_eq(fighter["active_affliction"], StressAfflictionEngine.AfflictionType.GUN_SHY, "Trauma must cause GUN-SHY")
    assert_eq(fighter["stress"], 60.0, "Stress must reset to 60 on cathartic break")

func test_gun_shy_reduces_power_and_fight_iq_in_bout():
    var fighter = {"active_affliction": StressAfflictionEngine.AfflictionType.GUN_SHY}
    var base_stats = {"power": 75, "fight_iq": 60, "defense": 70, "conditioning": 80}
    var modded = StressAfflictionEngine.apply_fight_night_stat_modifiers(fighter, base_stats)
    
    assert_eq(modded["power"], 65, "GUN-SHY must subtract 10 power")
    assert_eq(modded["fight_iq"], 55, "GUN-SHY must subtract 5 fight_iq")
    assert_eq(modded["defense"], 70, "Unaffected stats must remain unchanged")

Footwork & Ring Craft (boxing_pivot_footwork, boxing_angle_step, etc.): Top-down ring canvas grid view showing footprint decals with directional arc arrows demonstrating angle cuts ($45^\circ$ / $90^\circ$ pivot).
Combinations (boxing_one_two_hook, boxing_liver_combo, etc.): Sequential layered fist silhouettes numbered 1 -> 2 -> 3 connected by flowing energy ribbons.
Infighting (boxing_clinch_tieup, boxing_inside_uppercut, etc.): Collar-and-elbow close-contact chest silhouettes with forearm wedge lines and short-range vertical uppercut arrows.

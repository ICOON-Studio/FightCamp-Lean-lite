Missing Attribute Detail Breakdowns: The table only shows high-level meta (RECORD, AGE, DIVISION, POTENTIAL, ASK, FEE). A PC sports management title (Football Manager / OOTP class) requires scouted ranges for core pillars: Power, Speed, Defense, Conditioning, Fight IQ, Chin, and Style.

1. Fog-of-War Attribute Bar Component (fog_attribute_bar.gd)
class_name FogAttributeBar
extends Control

@export var attribute_name: String = "Power"
@export var min_val: int = 50:
	set(v):
		min_val = clampi(v, 0, 100)
		queue_redraw()
@export var max_val: int = 75:
	set(v):
		max_val = clampi(v, min_val, 100)
		queue_redraw()
@export var is_revealed: bool = false:
	set(v):
		is_revealed = v
		queue_redraw()

# Round-70 Color Tokens
const COLOR_VOID: Color = Color("#0d0c0b")
const COLOR_TRACK: Color = Color("#22201d")
const COLOR_BORDER: Color = Color("#3d372e")
const COLOR_SOLID: Color = Color("#f0c233")
const COLOR_UNCERTAIN: Color = Color("#6e624f")
const COLOR_TEXT_CREAM: Color = Color("#f5eedc")
const COLOR_TEXT_MUTED: Color = Color("#9e9484")

func _draw() -> void:
	var total_w: float = size.x
	var total_h: float = size.y
	var bar_h: float = 12.0
	var bar_y: float = (total_h - bar_h) * 0.5
	var label_w: float = 110.0
	var bar_w: float = total_w - label_w - 70.0
	var bar_x: float = label_w
	
	# 1. Attribute Label
	draw_string(
		ThemeDB.fallback_font,
		Vector2(0, bar_y + 10.0),
		attribute_name.to_upper(),
		HORIZONTAL_ALIGNMENT_LEFT,
		label_w - 8.0,
		11,
		COLOR_TEXT_MUTED
	)
	
	# 2. Track Background
	var track_rect: Rect2 = Rect2(bar_x, bar_y, bar_w, bar_h)
	draw_rect(track_rect, COLOR_TRACK, true)
	draw_rect(track_rect, COLOR_BORDER, false, 1.0)
	
	# 3. Bar Fills
	var px_per_pt: float = bar_w / 100.0
	if is_revealed or min_val == max_val:
		var solid_w: float = float(min_val) * px_per_pt
		draw_rect(Rect2(bar_x, bar_y, solid_w, bar_h), COLOR_SOLID, true)
	else:
		var solid_w: float = float(min_val) * px_per_pt
		var range_w: float = float(max_val - min_val) * px_per_pt
		
		# Solid minimum confirmed capability
		if solid_w > 0.0:
			draw_rect(Rect2(bar_x, bar_y, solid_w, bar_h), COLOR_SOLID, true)
		
		# Uncertain potential zone
		var uncertain_rect: Rect2 = Rect2(bar_x + solid_w, bar_y, range_w, bar_h)
		draw_rect(uncertain_rect, COLOR_UNCERTAIN, true)
		
		# Diagonal Hatching over uncertain zone
		var hatch_spacing: float = 4.0
		var hatch_x: float = bar_x + solid_w
		while hatch_x < (bar_x + solid_w + range_w):
			draw_line(
				Vector2(hatch_x, bar_y + bar_h),
				Vector2(minf(hatch_x + 4.0, bar_x + solid_w + range_w), bar_y),
				COLOR_VOID,
				1.0
			)
			hatch_x += hatch_spacing

	# 4. Monospace Value Readout
	var text_val: String = str(min_val) if (is_revealed or min_val == max_val) else "%d-%d" % [min_val, max_val]
	draw_string(
		ThemeDB.fallback_font,
		Vector2(total_w - 60.0, bar_y + 10.0),
		text_val,
		HORIZONTAL_ALIGNMENT_RIGHT,
		60.0,
		11,
		COLOR_TEXT_CREAM
	)

2. Prospect List Row Component (prospect_row.gd)
class_name ProspectRow
extends PanelContainer

signal selected(prospect_data: Dictionary)
signal shortlist_toggled(prospect_id: String, is_shortlisted: bool)

# Round-70 Color & Style Tokens
const COLOR_BASE: Color = Color("#1a1816")
const COLOR_HOVER: Color = Color("#22201d")
const COLOR_SELECTED: Color = Color("#282521")
const COLOR_BORDER_DEFAULT: Color = Color("#3d372e")
const COLOR_BORDER_ACTIVE: Color = Color("#f0c233")
const COLOR_GOLD: Color = Color("#f0c233")
const COLOR_CREAM: Color = Color("#f5eedc")
const COLOR_MUTED: Color = Color("#9e9484")

var prospect_data: Dictionary = {}
var is_active_selection: bool = false
var is_shortlisted: bool = false

@onready var star_btn: Button = $HBox/StarBtn
@onready var name_lbl: Label = $HBox/IdentityBlock/NameLabel
@onready var sub_lbl: Label = $HBox/IdentityBlock/SubLabel
@onready var div_lbl: Label = $HBox/DivisionLabel
@onready var style_lbl: Label = $HBox/StyleLabel
@onready var record_lbl: Label = $HBox/RecordLabel
@onready var potential_lbl: Label = $HBox/PotentialLabel
@onready var wage_lbl: Label = $HBox/WageLabel
@onready var fee_lbl: Label = $HBox/FeeLabel
@onready var style_box: StyleBoxFlat = StyleBoxFlat.new()

func _ready() -> void:
	custom_minimum_size = Vector2(1380, 62)
	focus_mode = FocusMode.FOCUS_ALL
	
	style_box.corner_radius_top_left = 4
	style_box.corner_radius_top_right = 4
	style_box.corner_radius_bottom_left = 4
	style_box.corner_radius_bottom_right = 4
	style_box.bg_color = COLOR_BASE
	style_box.border_width_left = 1
	style_box.border_width_top = 1
	style_box.border_width_right = 1
	style_box.border_width_bottom = 1
	style_box.border_color = COLOR_BORDER_DEFAULT
	add_theme_stylebox_override("panel", style_box)
	
	mouse_entered.connect(_on_mouse_enter)
	mouse_exited.connect(_on_mouse_exit)
	gui_input.connect(_on_gui_input)
	if star_btn:
		star_btn.pressed.connect(_on_star_pressed)

func populate(data: Dictionary, selected_id: String) -> void:
	prospect_data = data
	is_active_selection = (data.get("id", "") == selected_id)
	is_shortlisted = data.get("is_shortlisted", false)
	
	name_lbl.text = data.get("name", "Unknown Boxer")
	sub_lbl.text = "%s · %d YO · %s" % [data.get("country", "UNK"), data.get("age", 18), data.get("stance", "Orthodox")]
	div_lbl.text = data.get("division", "Welterweight")
	style_lbl.text = data.get("style", "Boxer-Puncher")
	record_lbl.text = "%d-%d-%d (%d KO)" % [data.get("wins", 0), data.get("losses", 0), data.get("draws", 0), data.get("kos", 0)]
	
	var conf: float = data.get("scout_confidence", 0.3)
	if conf < 0.5:
		potential_lbl.text = "GRADE: B- ~ A"
		potential_lbl.modulate = COLOR_MUTED
	else:
		potential_lbl.text = "EST: %d - %d" % [data.get("pot_min", 70), data.get("pot_max", 85)]
		potential_lbl.modulate = COLOR_GOLD
		
	wage_lbl.text = "$%s/wk" % [String.num(data.get("wage_ask", 800), 0)]
	fee_lbl.text = "$%s" % [String.num(data.get("sign_fee", 2000), 0)]
	
	_update_star_visual()
	_update_panel_visual()

func _update_panel_visual() -> void:
	var tween: Tween = create_tween().set_parallel(true)
	if is_active_selection:
		tween.tween_property(style_box, "bg_color", COLOR_SELECTED, 0.08)
		tween.tween_property(style_box, "border_color", COLOR_BORDER_ACTIVE, 0.08)
	else:
		tween.tween_property(style_box, "border_color", COLOR_BORDER_DEFAULT, 0.08)
		tween.tween_property(style_box, "bg_color", COLOR_BASE, 0.08)

func _on_mouse_enter() -> void:
	if not is_active_selection:
		var tween: Tween = create_tween()
		tween.tween_property(style_box, "bg_color", COLOR_HOVER, 0.08)

func _on_mouse_exit() -> void:
	if not is_active_selection:
		var tween: Tween = create_tween()
		tween.tween_property(style_box, "bg_color", COLOR_BASE, 0.08)

func _on_gui_input(event: InputEvent) -> void:
	if event is InputEventMouseButton and event.button_index == MOUSE_BUTTON_LEFT and event.pressed:
		selected.emit(prospect_data)

func _on_star_pressed() -> void:
	is_shortlisted = !is_shortlisted
	_update_star_visual()
	shortlist_toggled.emit(prospect_data.get("id", ""), is_shortlisted)

func _update_star_visual() -> void:
	if star_btn:
		star_btn.modulate = COLOR_GOLD if is_shortlisted else COLOR_MUTED

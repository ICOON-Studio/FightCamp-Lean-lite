3. Corrected Failure Modes 3 and 4 (Godot 4.7 Skeleton Updates)

In Godot 4.7, modifying bone transforms during regular process loops causes transform jitter and breaks spatial caching. Bone modifications must be queued and written via deferred overrides during the post-evaluation pass.

Corrected Failure Mode 3: Base-of-Support Balance via Deferred Skeleton Pass
# File: FightAliveStance.gd
extends Node

func _process_skeleton_deferred_balance(skeleton: Skeleton3D, max_lean_ratio: float = 0.85) -> void:
	if not skeleton:
		return
	var pelvis_idx: int = skeleton.find_bone("Hips")
	var l_foot_idx: int = skeleton.find_bone("LeftFoot")
	var r_foot_idx: int = skeleton.find_bone("RightFoot")
	if pelvis_idx == -1 or l_foot_idx == -1 or r_foot_idx == -1:
		return
		
	var lf_pos: Vector3 = skeleton.get_bone_global_pose(l_foot_idx).origin
	var rf_pos: Vector3 = skeleton.get_bone_global_pose(r_foot_idx).origin
	var p_trans: Transform3D = skeleton.get_bone_global_pose(pelvis_idx)
	
	var support_center: Vector3 = (lf_pos + rf_pos) * 0.5
	var support_span: float = (lf_pos - rf_pos).length() * 0.5 * max_lean_ratio
	
	var com_horizontal: Vector3 = Vector3(p_trans.origin.x, 0.0, p_trans.origin.z)
	var base_horizontal: Vector3 = Vector3(support_center.x, 0.0, support_center.z)
	var offset: Vector3 = com_horizontal - base_horizontal
	
	if offset.length() > support_span:
		var clamped_pos: Vector3 = base_horizontal + (offset.normalized() * support_span)
		p_trans.origin.x = clamped_pos.x
		p_trans.origin.z = clamped_pos.z
		# Deferred override ensures proper engine skeleton batching
		skeleton.set_bone_global_pose_override(pelvis_idx, p_trans, 0.90, true)

Corrected Failure Mode 4: Cervical Spine Kinematic Chain with Chin-Tuck Clamp
# File: FightAliveStance.gd
extends Node

func _process_skeleton_deferred_head_look(skeleton: Skeleton3D, target_aim_point: Vector3, chin_tuck_factor: float) -> void:
	if not skeleton:
		return
	var head_idx: int = skeleton.find_bone("Head")
	var neck_idx: int = skeleton.find_bone("Neck")
	var spine2_idx: int = skeleton.find_bone("Spine2")
	if head_idx == -1 or neck_idx == -1 or spine2_idx == -1:
		return

	var head_trans: Transform3D = skeleton.get_bone_global_pose(head_idx)
	var forward_dir: Vector3 = -head_trans.basis.z.normalized()
	var target_dir: Vector3 = (target_aim_point - head_trans.origin).normalized()
	
	# Clamp horizontal aim angle to biological limits (max 48 deg total cervical rotation)
	var full_rotation: Quaternion = Quaternion(forward_dir, target_dir)
	var clamped_rot: Quaternion = Quaternion.IDENTITY.slerp(full_rotation, 0.45)
	
	# Split rotation down the cervical-thoracic chain
	var sp2_rot: Quaternion = skeleton.get_bone_pose_rotation(spine2_idx)
	var neck_rot: Quaternion = skeleton.get_bone_pose_rotation(neck_idx)
	var head_rot: Quaternion = skeleton.get_bone_pose_rotation(head_idx)
	
	skeleton.set_bone_pose_rotation(spine2_idx, sp2_rot * clamped_rot.slerp(Quaternion.IDENTITY, 0.70))
	skeleton.set_bone_pose_rotation(neck_idx, neck_rot * clamped_rot.slerp(Quaternion.IDENTITY, 0.40))
	
	# Chin-tuck offset rotates head downward along local X-axis toward the sternum
	var chin_pitch: Quaternion = Quaternion(Vector3.RIGHT, deg_to_rad(14.0 * clampf(chin_tuck_factor, 0.0, 1.0)))
	skeleton.set_bone_pose_rotation(head_idx, (head_rot * clamped_rot * chin_pitch).normalized())

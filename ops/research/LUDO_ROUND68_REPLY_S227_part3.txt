func _build_attribute_pillar(pillar_title: String, attr_definitions: Array) -> void:
	var pillar_box := VBoxContainer.new()
	pillar_box.add_theme_constant_override("separation", 6)

	var header := Label.new()
	header.text = pillar_title
	header.add_theme_font_size_override("font_size", UIKit.FONT_SIZE_MICRO)
	header.add_theme_color_override("font_color", UIKit.COLOR_BRASS_ANTIQUE)
	pillar_box.add_child(header)

	var row := HBoxContainer.new()
	row.add_theme_constant_override("separation", 10)

	for item in attr_definitions:
		var key: String = item[0]
		var title: String = item[1]
		var val: int = item[2]
		var icon: String = item[3]

		var tile := AttributeTile.new(title, val, icon)
		row.add_child(tile)
		tiles[key] = tile

	pillar_box.add_child(row)
	col_attributes.add_child(pillar_box)

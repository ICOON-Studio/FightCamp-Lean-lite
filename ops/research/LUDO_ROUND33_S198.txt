# Complete Camera Controller Implementation (Godot 4.7.2 Forward+)
class_name GymCameraController extends Node3D

@export var pan_speed: float = 18.0
@export var zoom_smoothness: float = 8.5
@export var hall_bounds_min: Vector2 = Vector2(4.0, 3.0)   # X, Z min in metres
@export var hall_bounds_max: Vector2 = Vector2(26.0, 15.0) # X, Z max in metres

var current_stop: int = 1 # 0: Overview, 1: Tactical, 2: Athlete
var target_zoom_y: float = 16.5
var target_zoom_z: float = 26.2
var target_pitch: float = -36.5
var target_fov: float = 38.0

@onready var camera: Camera3D = $Camera3D

func _unhandled_input(event: InputEvent) -> void:
    if event.is_action_pressed("camera_zoom_in"):
        step_zoom(-1)
    elif event.is_action_pressed("camera_zoom_out"):
        step_zoom(1)

func step_zoom(direction: int) -> void:
    current_stop = clampi(current_stop + direction, 0, 2)
    match current_stop:
        0: # Overview
            target_zoom_y = 16.5
            target_zoom_z = 26.2
            target_pitch = -36.5
            target_fov = 38.0
        1: # Tactical Zone
            target_zoom_y = 9.2
            target_zoom_z = 18.4
            target_pitch = -32.0
            target_fov = 42.0
        2: # Athlete Inspect
            target_zoom_y = 3.2
            target_zoom_z = 7.5
            target_pitch = -22.0
            target_fov = 54.0

func _process(delta: float) -> void:
    # Smooth zoom interpolation
    camera.position.y = lerpf(camera.position.y, target_zoom_y, delta * zoom_smoothness)
    camera.position.z = lerpf(camera.position.z, target_zoom_z, delta * zoom_smoothness)
    camera.rotation_degrees.x = lerpf(camera.rotation_degrees.x, target_pitch, delta * zoom_smoothness)
    camera.fov = lerpf(camera.fov, target_fov, delta * zoom_smoothness)
    
    # Edge panning (only permitted in Stop 1 and Stop 2)
    if current_stop != 0:
        var pan_input: Vector2 = Input.get_vector("pan_left", "pan_right", "pan_up", "pan_down")
        position.x = clampf(position.x + pan_input.x * pan_speed * delta, hall_bounds_min.x, hall_bounds_max.x)
        position.z = clampf(position.z + pan_input.y * pan_speed * delta, hall_bounds_min.y, hall_bounds_max.y)
    else:
        # Snap back to center hall on overview
        position.x = lerpf(position.x, 15.0, delta * 4.0)
        position.z = lerpf(position.z, 9.0, delta * 4.0)

# Zone System Core Mechanics Resource (Godot 4.7.2)
class_name GymZone extends Resource

@export var zone_id: String                  # e.g., "zone_bag_line"
@export var zone_name: String                # "HEAVY BAG LINE"
@export var base_capacity: int = 6           # Max concurrent athletes
@export var zone_tier: int = 1               # T1, T2, T3
@export var assigned_coach: Coach = null    # Coach stationed here
@export var hardware_condition: float = 1.0  # 1.0 = 100% pristine, 0.0 = broken

# 1. Congestion Modifier
# When athletes > capacity, training efficiency drops and collision/temper events trigger
func get_congestion_penalty(current_occupants: int) -> float:
    if current_occupants <= base_capacity:
        return 1.0 # 100% full efficiency
    var overflow: int = current_occupants - base_capacity
    # Every athlete over capacity drops zone attribute yield by 18% for everyone in it
    return clampf(1.0 - (overflow * 0.18), 0.28, 1.0)

# 2. Zone Tier Hardware Multiplier
# Higher tier zones provide better ergonomic support, reducing baseline wear
func get_wear_mitigation_factor() -> float:
    match zone_tier:
        1: return 1.00 # Standard wear
        2: return 0.82 # -18% wear (Commercial grade fluid-filled bags, high-density foam)
        3: return 0.65 # -35% wear (Olympic spec hydraulic platforms, force-plate telemetry)
    return 1.00

# 3. Assigned Coach Aura
# A coach stationed in a zone provides active real-time corrections to all occupants
func get_coaching_boost() -> Dictionary:
    if assigned_coach == null:
        return {"xp_mult": 1.0, "injury_mod": 1.0}
    return {
        "xp_mult": 1.0 + (assigned_coach.specialist_rating * 0.005), # e.g., 80 stat = +40% XP
        "injury_mod": clampf(1.0 - (assigned_coach.safety_iq * 0.006), 0.20, 1.0) # Halves injury risk
    }

@export var target_skeleton: Skeleton3D
@export var anim_tree: AnimationTree

var samples_recorded: int = 0
var max_samples: int = 360 # 6 seconds at 60 Hz
var root_accel_spikes: int = 0
var foot_slide_events: int = 0
var penetration_events: int = 0

var last_pelvis_vel: Vector3 = Vector3.ZERO
var lead_ankle_idx: int = -1
var pelvis_idx: int = -1
var head_idx: int = -1
var lead_wrist_idx: int = -1

func _ready() -> void:
    assert(target_skeleton != null, "Target skeleton must be assigned")
    pelvis_idx = target_skeleton.find_bone("Hips")
    lead_ankle_idx = target_skeleton.find_bone("LeftFoot")
    head_idx = target_skeleton.find_bone("Head")
    lead_wrist_idx = target_skeleton.find_bone("LeftHand")

func _physics_process(delta: float) -> void:
    if samples_recorded >= max_samples:
        return
    samples_recorded += 1

    # 1. Root & Pelvis Acceleration Measurement
    var current_pelvis_pos: Vector3 = target_skeleton.get_bone_global_pose(pelvis_idx).origin
    var current_pelvis_vel: Vector3 = (current_pelvis_pos - target_skeleton.get_bone_pose_position(pelvis_idx)) / delta
    var pelvis_accel: float = ((current_pelvis_vel - last_pelvis_vel) / delta).length()
    last_pelvis_vel = current_pelvis_vel
    if pelvis_accel > 25.0:
        root_accel_spikes += 1

    # 2. Planted Foot Slip Measurement (sample during stance contact phases)
    var foot_ground_weight: float = anim_tree.get("parameters/conditions/lead_foot_planted") if anim_tree else 1.0
    if foot_ground_weight > 0.5:
        var ankle_pos: Vector3 = target_skeleton.get_bone_global_pose(lead_ankle_idx).origin
        var planar_displacement: float = Vector2(ankle_pos.x, ankle_pos.z).length()
        if planar_displacement > 0.012: # 12 mm threshold
            foot_slide_events += 1

    # 3. Guard Self-Intersection (Head vs Lead Hand)
    var head_pos: Vector3 = target_skeleton.get_bone_global_pose(head_idx).origin
    var wrist_pos: Vector3 = target_skeleton.get_bone_global_pose(lead_wrist_idx).origin
    var glove_to_head_dist: float = head_pos.distance_to(wrist_pos)
    var combined_radii: float = 0.095 + 0.090 # 185 mm
    if glove_to_head_dist < (combined_radii - 0.012):
        penetration_events += 1

    if samples_recorded >= max_samples:
        var report: Dictionary = {
            "root_spikes": root_accel_spikes,
            "foot_slides": foot_slide_events,
            "penetrations": penetration_events,
            "status": "PASS" if (root_accel_spikes == 0 and foot_slide_events < 5 and penetration_events == 0) else "FAIL"
        }
        diagnostic_completed.emit(report)

Canonical Fatigue, Snap, and Readiness Formulas
# Canonical Fight Week Physics & Physiology Formulas
class_name FightWeekPhysiology extends RefCounted

# 1. Net Daily Stamina Burn Equation
static func compute_hourly_fatigue(intensity: float, base_conditioning: float, gym_temp_celsius: float) -> float:
    # intensity: 0.0 (Rest) to 3.0 (Max Spar)
    # base_conditioning: 1 to 100
    var temp_penalty: float = 1.0 + maxf(0.0, (gym_temp_celsius - 22.0) * 0.035)
    var burn_rate: float = (intensity * 8.5) / (1.0 + (base_conditioning * 0.012))
    return burn_rate * temp_penalty

# 2. Punch Snap Decay & Peak Equation
# Snap measures neuromuscular explosive discharge. Too much fatigue kills snap; too much rest makes snap sluggish.
static func compute_snap(base_snap: float, stamina_pct: float, hours_since_high_intensity: float) -> float:
    var freshness_curve: float = 1.0
    if hours_since_high_intensity < 18.0:
        # Neuromuscular post-activation potentiation peak occurs between 18 and 36 hours post-spar
        freshness_curve = 0.88 + (hours_since_high_intensity / 18.0) * 0.12
    elif hours_since_high_intensity > 48.0:
        # Sluggishness decay kicks in if idle > 48 hours
        freshness_curve = maxf(0.75, 1.0 - ((hours_since_high_intensity - 48.0) * 0.008))
    
    var stamina_factor: float = clampf(stamina_pct / 80.0, 0.5, 1.0)
    return base_snap * freshness_curve * stamina_factor

# 3. Composite Fight Readiness Rating (0 to 100%)
static func compute_fight_readiness(stamina: float, snap: float, max_tissue_wear: float, weight_miss_lbs: float) -> float:
    var stamina_component: float = clampf(stamina, 0.0, 100.0) * 0.35
    var snap_component: float = clampf(snap, 0.0, 100.0) * 0.40
    var wear_penalty: float = max_tissue_wear * 0.25 # Highest single damaged body part
    var weight_penalty: float = 0.0
    if weight_miss_lbs > 0.0:
        weight_penalty = 25.0 + (weight_miss_lbs * 15.0) # Massive penalty if over limit
    
    return clampf((stamina_component + snap_component) - wear_penalty - weight_penalty, 0.0, 100.0)

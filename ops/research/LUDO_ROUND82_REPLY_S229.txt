func show_day(day_number: int, fighter_name: String, lived: Dictionary) -> void:
	# Parse lived dictionary via static helpers
	var top_gains: Array = compute_top_gains(lived.get("gains", {}), 3)
	var time_stats: Dictionary = compute_training_stats(lived)
	var coach_line: String = evaluate_coach_line(lived, time_stats)
	var chip_data: Array = compute_station_chips(lived.get("plan", []))

	# Update Display
	_title_label.text = "DAY %d WRAP - %s" % [day_number, fighter_name.to_upper()]
	_coach_line_label.text = coach_line

	var cur_stamina: float = float(lived.get("stamina", 100.0))
	_stats_summary_label.text = "Trained: %dm  |  Idle: %dm  |  With Coach: %dm  |  Stamina: %d%%" % [
		time_stats.trained_minutes,
		time_stats.idle_minutes,
		time_stats.coach_minutes,
		int(round(cur_stamina))
	]

	# Build gains badges
	for child in _gains_box.get_children():
		child.queue_free()

	if top_gains.is_empty():
		var none_label: Label = Label.new()
		none_label.text = "No gains recorded"
		none_label.add_theme_color_override("font_color", COLOR_MUTED)
		_gains_box.add_child(none_label)
	else:
		for gain in top_gains:
			var tag_panel: PanelContainer = PanelContainer.new()
			var tag_style: StyleBoxFlat = StyleBoxFlat.new()
			tag_style.bg_color = COLOR_MUTED_BG
			tag_style.border_color = COLOR_GOLD
			tag_style.border_width_left = 1
			tag_style.border_width_top = 1
			tag_style.border_width_right = 1
			tag_style.border_width_bottom = 1
			tag_style.corner_radius_top_left = 3
			tag_style.corner_radius_top_right = 3
			tag_style.corner_radius_bottom_left = 3
			tag_style.corner_radius_bottom_right = 3
			tag_style.content_margin_left = 6
			tag_style.content_margin_right = 6
			tag_style.content_margin_top = 3
			tag_style.content_margin_bottom = 3
			tag_panel.add_theme_stylebox_override("panel", tag_style)

			var tag_label: Label = Label.new()
			var stat_key: String = gain.stat
			var disp_name: String = STAT_NAMES.get(stat_key, stat_key.to_upper())
			tag_label.text = "+%.1f %s" % [gain.amount, disp_name]
			tag_label.add_theme_color_override("font_color", COLOR_GOLD)
			tag_panel.add_child(tag_label)
			_gains_box.add_child(tag_panel)

	# Build station chips
	for child in _station_chips_box.get_children():
		child.queue_free()

	for c in chip_data:
		var chip: PanelContainer = PanelContainer.new()
		var c_style: StyleBoxFlat = StyleBoxFlat.new()
		c_style.bg_color = COLOR_PANEL_ALT
		c_style.border_color = COLOR_PANEL_BORDER
		c_style.border_width_left = 1
		c_style.border_width_top = 1
		c_style.border_width_right = 1
		c_style.border_width_bottom = 1
		c_style.corner_radius_top_left = 4
		c_style.corner_radius_top_right = 4
		c_style.corner_radius_bottom_left = 4
		c_style.corner_radius_bottom_right = 4
		c_style.content_margin_left = 8
		c_style.content_margin_right = 8
		c_style.content_margin_top = 2
		c_style.content_margin_bottom = 2
		chip.add_theme_stylebox_override("panel", c_style)

		var chip_label: Label = Label.new()
		chip_label.text = "%s: %dm" % [c.name, c.minutes]
		chip_label.add_theme_color_override("font_color", COLOR_CREAM)
		chip.add_child(chip_label)
		_station_chips_box.add_child(chip)

	# Delayed entrance: slide up 2.5 seconds after day ends
	if _anim_tween and _anim_tween.is_valid():
		_anim_tween.kill()

	_anim_tween = create_tween()
	_anim_tween.tween_interval(2.5)
	_anim_tween.tween_callback(Callable(self, "_animate_slide_up"))

func _animate_slide_up() -> void:
	_drawer_panel.visible = true
	var h: float = _drawer_panel.size.y
	if h <= 0:
		h = _drawer_panel.custom_minimum_size.y
	_drawer_panel.offset_top = 0
	_drawer_panel.offset_bottom = h

	var tween: Tween = create_tween().set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_OUT)
	tween.tween_property(_drawer_panel, "offset_top", -h, 0.35)
	tween.parallel().tween_property(_drawer_panel, "offset_bottom", 0.0, 0.35)

# --- Pure Testable Calculation Helpers ---

static func compute_top_gains(gains: Dictionary, count: int = 3) -> Array:
	var list: Array = []
	for stat in gains.keys():
		var amt: float = float(gains[stat])
		if amt > 0.0001:
			list.append({"stat": str(stat), "amount": amt})

	list.sort_custom(func(a, b): return a.amount > b.amount)

	if list.size() > count:
		return list.slice(0, count)
	return list

static func compute_training_stats(lived: Dictionary) -> Dictionary:
	var plan: Array = lived.get("plan", [])
	var trained: int = 0
	var idle: int = 0
	var coach_mins: int = lived.get("coach_minutes", 0)

	for block in plan:
		if block.has("idle"):
			idle += int(block.get("idle", 0))
		elif block.has("station"):
			var m: int = int(block.get("minutes", 0))
			trained += m
			# Fallback if coach_minutes was not explicitly summarized at root
			if not lived.has("coach_minutes") and block.get("coach", false):
				coach_mins += m

	return {
		"trained_minutes": trained,
		"idle_minutes": idle,
		"total_minutes": trained + idle,
		"coach_minutes": coach_mins
	}

static func compute_station_chips(plan: Array) -> Array:
	var chips: Array = []
	var visited_order: Array = []
	var duration_by_kind: Dictionary = {}

	for block in plan:
		if block.has("station"):
			var kind: String = str(block.station)
			var minutes: int = int(block.get("minutes", 0))
			if not duration_by_kind.has(kind):
				visited_order.append(kind)
				duration_by_kind[kind] = 0
			duration_by_kind[kind] += minutes

	for kind in visited_order:
		var raw_name: String = STATION_LABELS.get(kind, kind.capitalize())
		chips.append({
			"kind": kind,
			"name": raw_name,
			"minutes": duration_by_kind[kind]
		})

	return chips

static func evaluate_coach_line(lived: Dictionary, stats: Dictionary) -> String:
	var strain: float = float(lived.get("strain", 0.0))
	var stamina: float = float(lived.get("stamina", 100.0))
	var idle_mins: int = stats.get("idle_minutes", 0)
	var trained_mins: int = stats.get("trained_minutes", 0)
	var coach_mins: int = stats.get("coach_minutes", 0)

	if strain >= 85.0 or stamina <= 15.0:
		return "He's running on fumes. Take it easy tomorrow or we break him."
	if idle_mins >= 40 or (trained_mins > 0 and float(idle_mins) / float(trained_mins + idle_mins) >= 0.40):
		return "Too much standing around today. We need tighter discipline."
	if coach_mins >= 45:
		return "Good day on the pads together. The technique is sharpening up."
	if stats.get("total_minutes", 0) == 0:
		return "No work put in today."
	if coach_mins == 0 and trained_mins > 30:
		return "He put the work in alone today, but he needs corner eyes on his mechanics."
	return "Solid shift today. Tomorrow we keep pushing."

func _test_top_gains_sorting() -> void:
	var gains: Dictionary = {
		"speed": 0.4,
		"power": 1.45,
		"technique": 0.1,
		"defense": 0.9,
		"chin": 0.0
	}
	var top3: Array = DayWrapDrawer.compute_top_gains(gains, 3)
	_assert(top3.size() == 3, "Top gains should be clamped to 3")
	_assert(top3[0].stat == "power" and is_equal_approx(top3[0].amount, 1.45), "Top 1 should be power")
	_assert(top3[1].stat == "defense" and is_equal_approx(top3[1].amount, 0.9), "Top 2 should be defense")
	_assert(top3[2].stat == "speed" and is_equal_approx(top3[2].amount, 0.4), "Top 3 should be speed")

func _test_stats_calculation() -> void:
	var lived: Dictionary = {
		"plan": [
			{"station": "pads", "minutes": 30, "coach": true},
			{"idle": 15, "why": "resting"},
			{"station": "heavy_bag", "minutes": 45, "coach": false},
			{"idle": 10, "why": "chatting"}
		],
		"coach_minutes": 30
	}
	var stats: Dictionary = DayWrapDrawer.compute_training_stats(lived)
	_assert(stats.trained_minutes == 75, "Trained minutes should sum station blocks (30+45=75)")
	_assert(stats.idle_minutes == 25, "Idle minutes should sum idle blocks (15+10=25)")
	_assert(stats.total_minutes == 100, "Total minutes should be 100")
	_assert(stats.coach_minutes == 30, "Coach minutes should equal 30")

def data(name):
    return bv.get_behaviour_data(bv.get_behaviour_by_name(objects[name], 'Transform'), 'global_position')


def position(name, frame=0):
    return np.array(dv.get_data_value(data(name), frame), dtype=float)


def unit(v):
    norm = np.linalg.norm(v)
    return v / norm if norm > 1e-8 else v


def smooth(t):
    t = max(0.0, min(1.0, t))
    return t * t * (3.0 - 2.0 * t)


def yaw(degrees):
    a = math.radians(degrees)
    c, s = math.cos(a), math.sin(a)
    return np.array([[c, 0.0, s], [0.0, 1.0, 0.0], [-s, 0.0, c]])


def elbow_target(shoulder, wrist, side, mode='hook'):
    upper, lower = lengths[side]
    delta = wrist - shoulder
    distance = np.linalg.norm(delta)
    max_reach = upper + lower - 0.5
    if distance >= max_reach:
        distance = max_reach
        delta = unit(delta) * distance
        wrist = shoulder + delta
    assert abs(upper - lower) < distance < upper + lower, (MOVE, side, distance, upper, lower)
    axis = delta / distance
    along = (upper * upper - lower * lower + distance * distance) / (2.0 * distance)
    h = math.sqrt(max(0.0, upper * upper - along * along))

    if mode == 'uppercut':
        # Uppercut pole drives downward under the arm line
        candidate = np.array([0.0, -1.0, 0.0])
    else:
        # Hook pole wings horizontally outward (left for left hand, +X)
        candidate = np.array([1.0, 0.25, 0.0]) if side == 'l' else np.array([-1.0, 0.25, 0.0])

    normal = candidate - axis * np.dot(candidate, axis)
    if np.linalg.norm(normal) < 1e-4:
        normal = np.array([0.0, 1.0, 0.0]) - axis * np.dot(np.array([0.0, 1.0, 0.0]), axis)
    pole = unit(normal)
    return shoulder + axis * along + pole * h


lengths = {s: (np.linalg.norm(position('upperarm_' + s) - position('lowerarm_' + s)),

    else:

    lat_shift = 4.0 * strike if MOVE == 'lead_hook' else 2.5 * strike

    turn_deg = (30.0 if MOVE == 'lead_hook' else 22.0) * strike

    # Move spine, pelvis, neck, and shoulders with core turn and forward weight shift
    for n, p in guard.items():

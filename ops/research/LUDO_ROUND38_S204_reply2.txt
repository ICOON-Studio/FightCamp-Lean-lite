Let $S \in [0.0, 1.0]$ be normalized current stamina, and let $R \in [0.0, 1.0]$ be cumulative bout progression (round elapsed fraction over total scheduled distance).

The effective exhaustion index $E \in [0.0, 1.0]$ is governed by an exponential onset curve with stamina threshold $S_{\text{crit}} = 0.40$:
$$E(S, R) = \begin{cases}
0.0 & \text{if } S \ge S_{\text{crit}} \
\left(\frac{S_{\text{crit}} - S}{S_{\text{crit}}}\right)^{1.65} \cdot \left(0.70 + 0.30 \cdot R\right) & \text{if } S < S_{\text{crit}}
\end{cases}$$

Individual postural displacement channels scale off $E$:

Wrist Vertical Sag ($D_{\text{wrist}}$): $$D_{\text{wrist}} = E \cdot D_{\text{wrist,max}} \quad \text{where } D_{\text{wrist,max}} = 0.12\text{ m (12.0 cm, nominal center of 8–14 cm spec)}.$$
Head Forward Projection ($Z_{\text{head}}$): $$Z_{\text{head}} = E \cdot 0.052\text{ m (5.2 cm)}.$$
Thoracic Pitch ($\theta_{\text{kyphosis}}$): $$\theta_{\text{kyphosis}} = E \cdot 0.113\text{ rad (6.5 degrees forward pitch around local X-axis)}.$$
Hyperventilation Cycle ($H(t)$): $$f_{\text{breath}} = 0.25 + 0.90 \cdot E \quad (\text{Hz, from } 15\text{ breaths/min to } 69\text{ breaths/min})$$ $$A_{\text{breath}} = E \cdot 0.038\text{ rad (2.2 degrees chest rise)}$$ $$\theta_{\text{breath}}(t) = A_{\text{breath}} \cdot \sin(2\pi f_{\text{breath}} t)$$

5. STEP 9 — GLOVE AND SECONDARY MOTION DYNAMICS
5.1 Physics Formulation: Semi-Implicit Underdamped Spring-Damper

Secondary recoil applies exclusively to extremities upon distinct excitation triggers:

Strike Hard Stop (Whiff at full extension): Glove overshoots, oscillations dampen.
Impact Received (Glove blocks punch): Sudden impulse compresses glove backward into guard line.
Contact Recoil: Clean punch rebound off target tissue.

The motion is governed by a second-order linear dynamic system:
$$m \ddot{\mathbf{x}} + c \dot{\mathbf{x}} + k \mathbf{x} = \mathbf{F}_{\text{ext}}(t)$$
Where:

$\mathbf{x}$ is the displacement vector from the base animated bone pose.
$\omega_0 = \sqrt{k/m}$ is the undamped natural frequency.
$\zeta = \frac{c}{2\sqrt{km}}$ is the damping ratio.
For punch stop/rebound, we set $\zeta = 0.42$ (underdamped: $1.5$ oscillations before rest).
For impact received, we set $\zeta = 0.72$ (approaching critical damping: quick snap-back, zero oscillation).
Benchmark requirement: Rebound displacement magnitude $|\mathbf{x}|$ must peak between $1.0\text{ cm}$ and $1.8\text{ cm}$ ($0.010 - 0.018\text{ m}$).

class_name GloveSpringModifier
extends SkeletonModifier3D

@export var enable_glove_springs: bool = false # HARD SWITCH (Rule A)
@export var measured_rebound_cm: float = 0.0 # Telemetry instrument

# Real-world calibrated constants
const SPRING_STIFFNESS_K: float = 480.0 # N/m (Glove cuff + wrist tendon stiffness)
const EFFECTIVE_MASS_M: float = 0.85 # kg (Hand + 10oz glove mass)
const DAMPING_UNDERDAMPED_C: float = 17.0 # zeta ~ 0.42
const DAMPING_BLOCK_C: float = 29.1 # zeta ~ 0.72

class SpringChannel:
	var bone_index: int = -1
	var position_offset: Vector3 = Vector3.ZERO
	var velocity: Vector3 = Vector3.ZERO
	var target_offset: Vector3 = Vector3.ZERO
	var damping_c: float = 17.0

var _skeleton: Skeleton3D = null
var _left_channel: SpringChannel = SpringChannel.new()
var _right_channel: SpringChannel = SpringChannel.new()
var _is_initialized: bool = false
var _contact_window_locked: bool = false

func _init() -> void:
	active = true

func set_contact_window_locked(locked: bool) -> void:
	_contact_window_locked = locked
	if locked:
		# Reset spring offsets instantly during certified contact window (Rule B)
		_left_channel.position_offset = Vector3.ZERO
		_left_channel.velocity = Vector3.ZERO
		_right_channel.position_offset = Vector3.ZERO
		_right_channel.velocity = Vector3.ZERO

func trigger_impulse(is_right_hand: bool, impulse_newton_seconds: Vector3, is_block: bool = false) -> void:
	if not enable_glove_springs or _contact_window_locked:
		return
	var chan: SpringChannel = _right_channel if is_right_hand else _left_channel
	chan.damping_c = DAMPING_BLOCK_C if is_block else DAMPING_UNDERDAMPED_C
	# delta v = J / m
	chan.velocity += impulse_newton_seconds / EFFECTIVE_MASS_M

func _validate_setup() -> void:
	_skeleton = get_skeleton()
	if _skeleton == null:
		return
	_left_channel.bone_index = _skeleton.find_bone("mixamorig_LeftHand")
	_right_channel.bone_index = _skeleton.find_bone("mixamorig_RightHand")
	_is_initialized = (_left_channel.bone_index >= 0 and _right_channel.bone_index >= 0)

func _process_modification() -> void:
	if not enable_glove_springs:
		measured_rebound_cm = 0.0
		return

	if not _is_initialized:
		_validate_setup()
		if not _is_initialized:
			return

	if _contact_window_locked:
		measured_rebound_cm = 0.0
		return # Rule B Protection

	var dt: float = get_physics_process_delta_time()
	dt = clampf(dt, 0.001, 0.033)

	_integrate_spring(_left_channel, dt)
	_integrate_spring(_right_channel, dt)

	measured_rebound_cm = maxf(_left_channel.position_offset.length(), _right_channel.position_offset.length()) * 100.0

	# Apply additive offsets to bone local transforms
	if _left_channel.bone_index >= 0:
		var p: Vector3 = _skeleton.get_bone_pose_position(_left_channel.bone_index)
		_skeleton.set_bone_pose_position(_left_channel.bone_index, p + _left_channel.position_offset)

	if _right_channel.bone_index >= 0:
		var p: Vector3 = _skeleton.get_bone_pose_position(_right_channel.bone_index)
		_skeleton.set_bone_pose_position(_right_channel.bone_index, p + _right_channel.position_offset)

func _integrate_spring(chan: SpringChannel, dt: float) -> void:
	# Semi-implicit Euler integration
	var spring_force: Vector3 = -SPRING_STIFFNESS_K * (chan.position_offset - chan.target_offset)
	var damping_force: Vector3 = -chan.damping_c * chan.velocity
	var acceleration: Vector3 = (spring_force + damping_force) / EFFECTIVE_MASS_M

	chan.velocity += acceleration * dt
	chan.position_offset += chan.velocity * dt

	# Clamp maximum deflection to 3.0 cm for biological sanity
	if chan.position_offset.length() > 0.03:
		chan.position_offset = chan.position_offset.normalized() * 0.03

Mathematical Formulation

Boxer fatigue combines short-term physical exertion with long-term systemic wear:

                      [PUNCHES / FOOTWORK / SLIPS]
                                   │
                                   ▼
             ┌───────────────────────────────────────────┐
             │       ACUTE EXERTION ACCUMULATION         │
             │       tau_recov = 4.5s (Phosphagen)       │
             └─────────────────────┬─────────────────────┘
                                   │
                         Overflow Exceeds VO2max
                                   │
                                   ▼
             ┌───────────────────────────────────────────┐
             │      LACTATE / CARDIOVASCULAR FATIGUE     │
             │       tau_recov = 45.0s (Aerobic)         │
             └─────────────────────┬─────────────────────┘
                                   │
                         Cumulative Ring Mileage
                                   │
                                   ▼
             ┌───────────────────────────────────────────┐
             │         SYSTEMIC POSTURAL COLLAPSE        │
             │      tau_recov = 420.0s (Between Rounds)  │
             └───────────────────────────────────────────┘


Kinetic Work Calculation ($W_{\text{strike}}$):
$$W_{\text{strike}} = \frac{1}{2} M_{\text{arm}} v_{\text{glove}}^2 + E_{\text{trunk_rot}}$$
Where:

$M_{\text{arm}} = 3.85\text{ kg}$ (combined arm mass).
$v_{\text{glove}}$ ranges from $9.1\text{ m/s}$ (jab) to $12.5\text{ m/s}$ (cross).
$E_{\text{trunk_rot}} \approx 45.0\text{ J}$ (metabolic work done by core muscles).

Acute Stamina Depletion Dynamics:
$$\frac{dS_{\text{acute}}}{dt} = -\frac{W_{\text{strike}}}{\text{Stamina Capacity}} + \frac{1 - S_{\text{acute}}}{\tau_{\text{fast}}}$$
Where $\tau_{\text{fast}} = 4.5\text{ s}$ represents creatine phosphate resynthesis during lulls in combat.

Cardiovascular / Lactic Drain ($S_{\text{cardio}}$):
When $S_{\text{acute}} < 0.35$, anaerobic glycolysis causes systemic fatigue:
$$\frac{dS_{\text{cardio}}}{dt} = -k_{\text{lactic}} \cdot (0.35 - S_{\text{acute}})^{1.4} + \frac{1 - S_{\text{cardio}}}{\tau_{\text{slow}}}$$
Where:

$k_{\text{lactic}} = 0.082\text{ s}^{-1}$.
$\tau_{\text{slow}} = 48.0\text{ s}$ (aerobic clearance rate).

Postural Degradation Channels (Section 4 Telemetry):

Wrist Drop ($D_{\text{wrist}}$): $$D_{\text{wrist}} = D_{\text{max}} \cdot \left(\frac{1 - S_{\text{cardio}}}{1.0}\right)^{1.85} \quad (D_{\text{max}} = 0.14\text{ m})$$
Forward Cervical Extension ($Z_{\text{head}}$): $$Z_{\text{head}} = 0.055\text{ m} \cdot (1.0 - S_{\text{cardio}})^{1.2}$$
Respiratory Chest Amplitude ($A_{\text{breath}}$): $$A_{\text{breath}} = 0.04\text{ rad} \cdot \left[1.0 + 2.5 \cdot (1.0 - S_{\text{cardio}})\right]$$
Production GDScript: res://scripts/sim/FatigueSubsystem.gd
class_name FatigueSubsystem
extends RefCounted

# HARD INVARIANT (Rule A): Module default disabled
@export var enable_fatigue_subsystem: bool = false

# Internal Simulation State
var stamina_acute: float = 1.0     # [0.0, 1.0] Fast recovery (Phosphagen)
var stamina_cardio: float = 1.0    # [0.0, 1.0] Slow recovery (Lactate clearance)
var stamina_systemic: float = 1.0  # [0.0, 1.0] Round-over-round floor
var round_progression: float = 0.0 # [0.0, 1.0] Bout distance elapsed

# Biomechanical Sourced Constants
const TAU_FAST_RECOVERY_SEC: float = 4.5     # CP resynthesis half-life
const TAU_SLOW_RECOVERY_SEC: float = 48.0    # Lactic acid clearance constant
const LACTIC_THRESHOLD: float = 0.35        # Anaerobic transition boundary
const K_LACTIC_DRAIN: float = 0.082         # Lactic conversion coefficient
const MAX_WRIST_DROP_METERS: float = 0.14   # 14.0 cm physical limit
const MIN_WRIST_DROP_METERS: float = 0.08   # 8.0 cm lower qualification boundary

func reset_state() -> void:
	stamina_acute = 1.0
	stamina_cardio = 1.0
	stamina_systemic = 1.0
	round_progression = 0.0

func apply_strike_work(glove_velocity_mps: float, trunk_engagement_joules: float, base_capacity_joules: float) -> void:
	if not enable_fatigue_subsystem:
		return
	
	# Kinetic work = 0.5 * m * v^2 + trunk_work
	var arm_mass_kg: float = 3.85
	var kinetic_energy: float = 0.5 * arm_mass_kg * pow(glove_velocity_mps, 2.0)
	var total_joules: float = kinetic_energy + trunk_engagement_joules
	
	var drain_ratio: float = total_joules / maxf(base_capacity_joules, 1000.0)
	stamina_acute = clampf(stamina_acute - drain_ratio, 0.0, 1.0)

func advance_simulation(delta_sec: float) -> void:
	if not enable_fatigue_subsystem:
		return

	# 1. Fast Acute Phosphagen Regeneration
	if stamina_acute < 1.0:
		var fast_regen_rate: float = (1.0 - stamina_acute) / TAU_FAST_RECOVERY_SEC
		stamina_acute = clampf(stamina_acute + fast_regen_rate * delta_sec, 0.0, 1.0)

	# 2. Slow Cardio Lactic Accumulation and Clearance
	if stamina_acute < LACTIC_THRESHOLD:
		var anaerobic_overdraft: float = LACTIC_THRESHOLD - stamina_acute
		var drain: float = K_LACTIC_DRAIN * pow(anaerobic_overdraft, 1.4) * delta_sec
		stamina_cardio = clampf(stamina_cardio - drain, 0.0, 1.0)
	else:
		# Aerobic recovery
		var cardio_regen_rate: float = (1.0 - stamina_cardio) / TAU_SLOW_RECOVERY_SEC
		stamina_cardio = clampf(stamina_cardio + cardio_regen_rate * delta_sec, 0.0, 1.0)

	# Cardio stamina limits the acute recovery ceiling
	stamina_acute = minf(stamina_acute, stamina_cardio)

func apply_between_round_recovery(recovery_quality_factor: float) -> void:
	if not enable_fatigue_subsystem:
		return
	
	# 60-second corner rest recovery: restore up to 45% of lost cardio reserves
	var rest_bonus: float = 0.45 * clampf(recovery_quality_factor, 0.5, 1.5)
	stamina_cardio = clampf(stamina_cardio + (1.0 - stamina_cardio) * rest_bonus, 0.0, 1.0)
	stamina_acute = stamina_cardio

func get_wrist_drop_cm() -> float:
	if not enable_fatigue_subsystem:
		return 0.0 # Rule A: Bit-identical zero output when OFF
	
	var exhaustion_factor: float = pow(1.0 - stamina_cardio, 1.85)
	var drop_m: float = exhaustion_factor * MAX_WRIST_DROP_METERS
	return drop_m * 100.0

func get_head_forward_cm() -> float:
	if not enable_fatigue_subsystem:
		return 0.0
	return (pow(1.0 - stamina_cardio, 1.2) * 0.055) * 100.0

func get_breathing_parameters() -> Vector2:
	# Returns Vector2(frequency_hz, amplitude_rad)
	if not enable_fatigue_subsystem:
		return Vector2(0.25, 0.0)
	
	var exhaustion: float = 1.0 - stamina_cardio
	var freq: float = 0.25 + (0.90 * exhaustion) # 0.25 Hz to 1.15 Hz
	var amp: float = 0.038 * (1.0 + 1.5 * exhaustion)
	return Vector2(freq, amp)

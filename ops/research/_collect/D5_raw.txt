// ─── MATH ────────────────────────────────────────────────────────────
function segmentData(p0, p1, q0, q1) {
  const u = new THREE.Vector3().subVectors(p1, p0);
  const v = new THREE.Vector3().subVectors(q1, q0);
  const w = new THREE.Vector3().subVectors(p0, q0);
  const a = u.dot(u), b = u.dot(v), c = v.dot(v);
  const d = u.dot(w), e = v.dot(w);
  const det = a*c - b*b; let s=0, t=0;
  if (det < 1e-10) {
    s=0; t = c>1e-10 ? (b*s+e)/c : 0; t=THREE.MathUtils.clamp(t,0,1);
  } else {
    s = (b*e - c*d)/det; t = (a*e - b*d)/det;
    s=THREE.MathUtils.clamp(s,0,1); t=THREE.MathUtils.clamp(t,0,1);
  }
  const pS = new THREE.Vector3().addScaledVector(u,s).add(p0);
  const qT = new THREE.Vector3().addScaledVector(v,t).add(q0);
  return { dist:pS.distanceTo(qT), pS, qT };
}
function surfaceGap(a,b) {
  const {dist} = segmentData(a.p0,a.p1, b.p0,b.p1);
  return dist - (a.r+b.r);
}
function gripPoint(cap, blend) {
  return new THREE.Vector3().lerpVectors(cap.p0, cap.p1, THREE.MathUtils.clamp(blend,0,1));
}

// ─── TWO-ANGLE LIMB BUILDER ─────────────────────────────────────────
// Elevation (X) then azimuth (Y) — preserves rest length exactly
function buildLimb(jnt, dirBase, elev, azim, r, name) {
  const len = Math.hypot(...dirBase);
  const dir = new THREE.Vector3(...dirBase).normalize();
  dir.applyQuaternion(new THREE.Quaternion().setFromAxisAngle(new THREE.Vector3(1,0,0), elev));
  dir.applyQuaternion(new THREE.Quaternion().setFromAxisAngle(new THREE.Vector3(0,1,0), azim));
  const p0 = jnt.clone();
  const p1 = jnt.clone().add(dir.multiplyScalar(len));
  return {p0,p1,r,name};
}

// ─── SPINE TRANSFORM ─────────────────────────────────────────────────
// Pelvis → thorax: two angles — flex/extend (X) + side bend (Z)
function buildBody(state) {
  const T = new THREE.Vector3(state.x||0, state.y||0, state.z||0);
  const yaw = state.yaw||0;
  const spine = state.spine || {flex:0, side:0};

  function rotPt(x,y,z) {
    const c=Math.cos(yaw), s=Math.sin(yaw);
    return new THREE.Vector3(c*x - s*z, y, s*x + c*z).add(T);
  }

  // Pelvis — rigid base
  const pDef = LIMBS.pelvis;
  const pelvis = {
    p0: rotPt(...pDef.p0), p1: rotPt(...pDef.p1), r: pDef.r, name:'pelvis'
  };

  // Waist = top of pelvis
  const waist = pelvis.p1.clone();

  // Thorax — hinged at waist: flex (X) then side-bend (Z)
  const tDef = LIMBS.thorax;
  const tDir = new THREE.Vector3(...tDef.dir).normalize();
  tDir.applyQuaternion(new THREE.Quaternion().setFromAxisAngle(new THREE.Vector3(1,0,0), spine.flex));
  tDir.applyQuaternion(new THREE.Quaternion().setFromAxisAngle(new THREE.Vector3(0,0,1), spine.side));
  const thorax = {
    p0: waist.clone(),
    p1: waist.clone().add(tDir.multiplyScalar(Math.hypot(...tDef.dir))),
    r: tDef.r, name:'thorax'
  };

  // Transform point from thorax-local to world
  function thoraxPt(x,y,z) {
    const l = new THREE.Vector3(x,y,z);
    const basisZ = new THREE.Vector3().subVectors(thorax.p1, thorax.p0).normalize();
    const basisX = new THREE.Vector3(0,1,0).cross(basisZ).normalize();
    const basisY = basisZ.clone().cross(basisX).normalize();
    return thorax.p0.clone().add(
      basisX.multiplyScalar(x).add(basisY.multiplyScalar(y)).add(basisZ.multiplyScalar(z))
    );
  }

  // Head — on top of thorax
  const hDef = LIMBS.head;
  const headJnt = thoraxPt(...hDef.jnt);
  const head = buildLimb(headJnt, hDef.dir, state.limbs?.head?.elev||0, 0, hDef.r, 'head');

  // Arms — from thorax
  const arms = {};
  for (const side of ['r','l']) {
    const def = LIMBS[side+'Arm'];
    const jnt = thoraxPt(...def.jnt);
    const elev = state.limbs?.[side+'Arm']?.elev||0;
    const azim = state.limbs?.[side+'Arm']?.azim||0;
    arms[side+'Arm'] = buildLimb(jnt, def.dir, elev, azim, def.r, side+'Arm');
  }

  // Legs — from pelvis
  const legs = {};
  for (const side of ['r','l']) {
    const def = LIMBS[side+'Leg'];
    const jnt = rotPt(...def.jnt);
    const elev = state.limbs?.[side+'Leg']?.elev||0;
    const azim = state.limbs?.[side+'Leg']?.azim||0;
    legs[side+'Leg'] = buildLimb(jnt, def.dir, elev, azim, def.r, side+'Leg');
  }

  // Hip + back
  const hipDef = LIMBS.rHip;
  const rHip = buildLimb(rotPt(...hipDef.jnt), hipDef.dir, state.limbs?.rHip?.elev||0, 0, hipDef.r, 'rHip');
  const backDef = LIMBS.back;
  const backJnt = thoraxPt(...backDef.jnt);
  const back = buildLimb(backJnt, backDef.dir, state.limbs?.back?.elev||0, 0, backDef.r, 'back');

  return { pelvis, thorax, head, ...arms, ...legs, rHip, back };
}

// ─── COST FUNCTION ──────────────────────────────────────────────────
function makeCostFn(pose) {
  return (fullState) => {
    const F1 = buildBody(fullState.F1);
    const F2 = buildBody(fullState.F2);
    let total = 0;

    // ALL-PAIRS PENETRATION — every capsule vs every capsule
    let worstPen = 0, penPair = '';
    for (const [n1,c1] of Object.entries(F1)) {
      for (const [n2,c2] of Object.entries(F2)) {
        const gap = surfaceGap(c1,c2);
        if (gap < 0) {
          const pen = -gap;
          total += pen*pen*900;
          if (pen > worstPen) { worstPen = pen; penPair = `F1.${n1} ↔ F2.${n2}`; }
        }
      }
    }

    // Declared contacts
    let worstTouch = 0, touchPair = '';
    for (const c of pose.contacts||[]) {
      const capA = c.a.startsWith('F1') ? F1[c.a.slice(3)] : F2[c.a.slice(3)];
      const capB = c.b.startsWith('F1') ? F1[c.b.slice(3)] : F2[c.b.slice(3)];
      const gap = surfaceGap(capA, capB);
      const err = Math.abs(gap - (c.off||0));
      total += err*err*1200;
      if (err > worstTouch) { worstTouch = err; touchPair = `${c.a}↔${c.b}`; }
    }

    // Grips
    let worstGrip = 0, gripPair = '';
    for (const g of pose.grips||[]) {
      const capFrom = g.from.startsWith('F1') ? F1[g.from.slice(3)] : F2[g.from.slice(3)];
      const capTo = g.to.startsWith('F1') ? F1[g.to.slice(3)] : F2[g.to.slice(3)];
      const handPos = capFrom.p1;
      const targetPos = gripPoint(capTo, g.blend);
      const gripErr = Math.abs(handPos.distanceTo(targetPos) - (g.off||0));
      total += gripErr*gripErr*1500;
      if (gripErr > worstGrip) { worstGrip = gripErr; gripPair = `${g.from}→${g.to}`; }
    }

    // Mat violation
    let worstMat = 0, matPair = '';
    for (const [bName,body] of [['F1',F1],['F2',F2]]) {
      for (const [lName,cap] of Object.entries(body)) {
        const low = Math.min(cap.p0.y, cap.p1.y) - cap.r;
        const v = MAT_SURFACE_Y - low;
        if (v > 0) {
          total += v*v*1200;
          if (v > worstMat) { worstMat = v; matPair = `${bName}.${lName}`; }
        }
      }
    }

    // Hold lift
    let worstHold = 0, holdPart = '';
    for (const [bName,parts] of Object.entries(pose.hold||{})) {
      const body = bName==='F1' ? F1 : F2;
      for (const p of parts) {
        const cap = body[p];
        const lowest = Math.min(cap.p0.y, cap.p1.y) - cap.r;
        const lift = Math.max(0, lowest - MAT_SURFACE_Y);
        if (lift > 0.001) {
          total += lift*lift*1000;
          if (lift > worstHold) { worstHold = lift; holdPart = `${bName}.${p}`; }
        }
      }
    }

    // Stature/foot check
    const stature = (b) => {
      const all = Object.values(b);
      const top = Math.max(...all.map(c=>Math.max(c.p0.y,c.p1.y)));
      const bot = Math.min(...all.map(c=>Math.min(c.p0.y,c.p1.y)));
      return {height:top-bot, footY:bot};
    };
    const s1 = stature(F1), s2 = stature(F2);

    return { cost:total, worstPen, penPair, worstTouch, touchPair,
             worstGrip, gripPair, worstMat, matPair, worstHold, holdPart,
             s1, s2 };
  };
}

// ─── SOLVER ──────────────────────────────────────────────────────────
function zeroLimbs() {
  return {
    head:{elev:0},
    rArm:{elev:0,azim:0}, lArm:{elev:0,azim:0},
    rLeg:{elev:0,azim:0}, lLeg:{elev:0,azim:0},
    rHip:{elev:0}, back:{elev:0}
  };
}
function zeroSpine() { return {flex:0, side:0}; }
function makeInitial(pose) {
  return {
    F1: { ...pose.init.F1, limbs:zeroLimbs(), spine:zeroSpine() },
    F2: { ...pose.init.F2, limbs:zeroLimbs(), spine:zeroSpine() }
  };
}
function cloneState(s) {
  return {
    F1: { ...s.F1, limbs:JSON.parse(JSON.stringify(s.F1.limbs)), spine:{...s.F1.spine} },
    F2: { ...s.F2, limbs:JSON.parse(JSON.stringify(s.F2.limbs)), spine:{...s.F2.spine} }
  };
}
function perturb(state, dof, delta) {
  const s = cloneState(state);
  const [a,b,c,d] = dof.split('.');
  if (d==='elev' || d==='azim') s[a][b][c][d] += delta;
  else if (c==='flex' || c==='side') s[a][b][c] += delta;
  else if (c) s[a][b][c] += delta;
  else s[a][b] += delta;
  return s;
}
function listDOFs(includeSpine) {
  const out = [];
  for (const b of ['F1','F2']) {
    for (const k of ['x','y','z','yaw']) out.push(`${b}.${k}`);
    if (includeSpine) out.push(`${b}.spine.flex`, `${b}.spine.side`);
    for (const side of ['r','l']) {
      out.push(`${b}.limbs.${side}Arm.elev`, `${b}.limbs.${side}Arm.azim`);
      out.push(`${b}.limbs.${side}Leg.elev`, `${b}.limbs.${side}Leg.azim`);
    }
  }
  return out;
}
function solve(pose, includeSpine, statusCb) {
  let state = makeInitial(pose);
  const costFn = makeCostFn(pose);
  let evals = 0;
  const best = (s) => { ++evals; return { ...costFn(s), state:s, evals }; };
  let curr = best(state);
  const dofs = listDOFs(includeSpine);
  const stepBase = includeSpine ? 0.06 : 0.06;
  for (let phase=1; phase<=3; phase++) {
    let step = stepBase;
    for (let cycle=0; cycle<12 && step>0.003; cycle++) {
      let improved = false;
      for (const d of dofs) {
        for (const sc of [1,0.4,0.15]) {
          for (const sign of [-1,1]) {
            const trial = best(perturb(curr.state, d, sign*sc*step));
            if (trial.cost < curr.cost) {
              curr = trial; improved = true;
              statusCb?.({...curr, phase:`P${phase}`});
              break;
            }
          }
        }
      }
      if (!improved) step *= 0.4;
      if (evals > 25000) break;
    }
  }
  return curr;
}

1. Coordinate System
Y = up, X = right, Z = forward/back

2. Spine Definition — The Waist Hinge
plaintext
Pelvis (lower torso):
  p0 = (0.00, 0.50, 0.00)  — body-local base
  p1 = (0.00, 0.68, 0.00)  — WAIST = hinge point
  Length = 0.18 m  — FIXED, never scaled

Thorax (upper torso):
  p0 = waist position (from pelvis)
  p1 = (0.00, 0.85, 0.00) in neutral
  Rest vector = (0, 0.17, 0)
  Length = 0.17 m  — FIXED, never scaled

Hinge: TWO stacked rotations applied to thorax direction vector:
  1. Flexion/Extension  — about thorax-local X axis
     Range: -π/3 to +π/3 rad  (~-60° to +60°)
     +flex = bend forward
  2. Side Bend           — about thorax-local Z axis
     Range: -π/4 to +π/4 rad  (~-45° to +45°)
     +side = bend to body's right

These are applied sequentially to the direction vector; segment endpoints
move accordingly. Pelvis does NOT rotate — only thorax bends.

Two angles per arm, applied to direction vector:
  1. Elevation  — about shoulder-local X
     Range: -π/2 to +π/2 rad
     +elev = raise arm forward/up
  2. Azimuth    — about global Y (after elevation)

     +azim = swing arm outward/inward

Order: elevation FIRST, then azimuth.

5. Order of Operations — Critical for Matching
plaintext
FOR EACH BODY:
1. Root transform: position (x,y,z) + yaw — apply to pelvis and hip joints
2. Pelvis: fixed from root — no bend
3. Spine: apply flex → then side bend to thorax direction vector
   → thorax p1 computed
4. Shoulder joint positions derived from thorax orientation
5. Arms: apply elevation → then azimuth to each arm direction vector
   → hand positions computed
6. Legs: elevation → azimuth from fixed hip joints

Spine FIRST, then arms. Shoulders sit on thorax — if arms rotated first then spine bent, hands would swing with spine and lose target. Order matters.

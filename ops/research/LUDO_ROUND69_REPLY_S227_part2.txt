			_flash_selection()
		elif event.button_index == MOUSE_BUTTON_RIGHT:

			focus_dropped.emit(day_index, current_focus)

			tw.tween_property(hover_overlay, "color:a", 0.0, 0.08)

func populate(p_fighter: Dictionary) -> void:
	fighter_data = p_fighter
	name_label.text = str(fighter_data.get("name", "UNKNOWN")).to_upper()
	var plan: Array = fighter_data.get("plan", [0, 0, 0, 0, 0, 1, 0])

	for i in range(mini(plan.size(), day_cells.size())):
		var focus: int = int(plan[i])
		day_cells[i].set_cell_state(focus, TrainingConstants.Intensity.MEDIUM, i == 0)

	_recalculate_metrics()

func apply_preset(preset_key: String) -> void:
	if not TrainingConstants.PRESETS.has(preset_key):
		return
	var template: Array = TrainingConstants.PRESETS[preset_key]
	for i in range(mini(template.size(), day_cells.size())):
		day_cells[i].set_cell_state(template[i], TrainingConstants.Intensity.MEDIUM, i == 0)
	_recalculate_metrics()
	plan_mutated.emit(fighter_data.get("id", ""))

func _on_cell_focus_changed(day_idx: int, focus: int) -> void:
	var plan: Array = fighter_data.get("plan", [0, 0, 0, 0, 0, 0, 0])
	if day_idx < plan.size():
		plan[day_idx] = focus
	_recalculate_metrics()
	plan_mutated.emit(fighter_data.get("id", ""))

func _recalculate_metrics() -> void:
	var work_count := 0
	var rec_count := 0
	var rest_count := 0

	for cell in day_cells:
		match cell.current_focus:
			TrainingConstants.Focus.REST:
				rest_count += 1
			TrainingConstants.Focus.RECOVERY:
				rec_count += 1
			_:
				work_count += 1

	load_summary_label.text = "WORK: %d  |  REC: %d  |  REST: %d" % [work_count, rec_count, rest_count]

	# Dynamic calculations
	var calc_readiness: int = clampi(100 - (work_count * 14) + (rest_count * 10) + (rec_count * 6), 15, 100)
	var calc_cond: int = clampi(50 + (work_count * 10) - (rest_count * 4), 20, 100)
	var calc_fatigue: int = work_count * 6

	readiness_bar.value = calc_readiness
	readiness_val_label.text = "%d%%" % calc_readiness
	cond_bar.value = calc_cond
	cond_val_label.text = "%d%%" % calc_cond

	if work_count >= 5:
		carryover_label.text = "FATIGUE CARRY-OVER: +%d%% (DANGEROUS OVERLOAD)" % calc_fatigue
		carryover_label.add_theme_color_override("font_color", UIKit.COLOR_OXBLOOD_BRIGHT)
		status_strip.color = UIKit.COLOR_OXBLOOD_BRIGHT
	elif work_count >= 3:
		carryover_label.text = "FATIGUE CARRY-OVER: +%d%% (OPTIMAL PROGRESSION)" % calc_fatigue
		carryover_label.add_theme_color_override("font_color", UIKit.COLOR_GOLD_BRIGHT)
		status_strip.color = Color("#2f855a")
	else:
		carryover_label.text = "FATIGUE CARRY-OVER: +%d%% (CONDITIONING DECAY)" % calc_fatigue
		carryover_label.add_theme_color_override("font_color", UIKit.COLOR_BRASS_DARK)
		status_strip.color = UIKit.COLOR_BRASS_ANTIQUE

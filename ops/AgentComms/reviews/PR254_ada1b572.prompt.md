+## Fill accuracy / evasion / chin into a COPY of `stats` when they are missing.
+## accuracy  = technique
+## evasion   = round(0.5 * speed + 0.5 * defense)
+## chin      = 50 + round((defense - 50) * 0.2), clamped 30..90
+## Missing source keys default to 50. Existing keys are never overwritten.
+static func derive_stats(stats: Dictionary) -> Dictionary:
+	var out: Dictionary = stats.duplicate(true)
+	if not out.has("accuracy"):
+		var technique: float = float(out.get("technique", 50.0))
+		out["accuracy"] = technique
+	if not out.has("evasion"):
+		var speed: float = float(out.get("speed", 50.0))
+		var defense: float = float(out.get("defense", 50.0))
+		out["evasion"] = round(0.5 * speed + 0.5 * defense)
+	if not out.has("chin"):
+		var defense: float = float(out.get("defense", 50.0))
+		var chin: float = 50.0 + round((defense - 50.0) * 0.2)
+		out["chin"] = clampi(int(chin), 30, 90)
+	return out
+

+## The chance the DEFENDER makes the punch miss completely (LUDO r114 shape).
+## Grows with evasion, falls with attacker accuracy.
+##   base = 0.22 * pow(2e / (e + a), 1.5)   (e = max(evasion, 1), a = max(accuracy, 1))
+##   then * (1 + 0.6 * (1 - distance_fit))   (a wrong-distance punch, fit 0.45, is easier to slip)
+##   then + 0.20 * telegraph                 (telegraph 0..1, how readable the punch was)
+##   clamp 0.03 .. 0.60
+## Calibration: 60/60 -> 0.22;  80/40 -> 0.339;  40/80 -> 0.120.
+static func whiff_chance(evasion: float, accuracy: float, distance_fit: float = 1.0, telegraph: float = 0.0) -> float:
+	var e: float = maxf(evasion, 1.0)
+	var a: float = maxf(accuracy, 1.0)
+	var base: float = 0.22 * pow(2.0 * e / (e + a), 1.5)
+	base *= 1.0 + 0.6 * (1.0 - distance_fit)
+	base += 0.20 * telegraph
+	return clampf(base, 0.03, 0.60)
+

+## What the thrower pays when he whiffs (LUDO r114): 1.5x the punch's stamina, and the
+## defender gets a 0.45 s counter window.
+static func whiff_cost() -> Dictionary:
+	return {"stamina_mult": 1.5, "counter_window_s": 0.45}
+
+
+## Multiplies the attacker's clean / glance thresholds (0-100). 50 -> 1.00, 100 -> 1.15, 0 -> 0.85.
+static func accuracy_mult(accuracy: float) -> float:
+	return 0.85 + 0.30 * float(accuracy) / 100.0
+
+
+## Multiplies head shock AND the knockdown chance (0-100). 50 -> 1.00, 80 -> 0.64, 30 -> 1.24.
+static func chin_mult(chin: float) -> float:
+	return clampf(1.6 - 1.2 * float(chin) / 100.0, 0.5, 1.4)
+
+
+## Today's engine: 0.10 * force^2 * (1 + head_wear), capped at 0.20; return that times
+## chin_mult(chin), capped at 0.20 * 1.4.
+static func head_shock(force: float, chin: float, head_wear: float) -> float:
+	var raw: float = 0.10 * force * force * (1.0 + head_wear)
+	var capped: float = minf(raw, 0.20)
+	return minf(capped * chin_mult(chin), 0.20 * 1.4)

+static func knockdown_chance(head_wear: float, chin: float) -> float:
+	var base: float = (0.09 + 0.12 * head_wear) * chin_mult(chin)
+	return clampf(base, 0.0, 0.6)
